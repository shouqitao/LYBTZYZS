### 1.2 技术特点
- **MVVM架构**：完全分离视图和业务逻辑
- **模块化设计**：每个功能模块独立可插拔
- **依赖注入**：使用DryIoc容器管理依赖
- **数据绑定**：利用WPF强大的数据绑定能力
- **响应式UI**：支持异步操作和加载状态
- **Material Design**：现代化的UI设计风格

#### 管理页面基类
```csharp
public abstract class BaseManagementViewModel<TEntity, TService> : ViewModelBase
    where TEntity : class
    where TService : class
{
    protected readonly TService Service;
    
    private ObservableCollection<TEntity> _items = new();
    private TEntity? _selectedItem;
    private string _searchKeyword = string.Empty;
    private int _currentPage = 1;
    private int _pageSize = 20;
    private int _totalCount;
    
    // 属性
    public ObservableCollection<TEntity> Items
    {
        get => _items;
        set => SetProperty(ref _items, value);
    }
    
    public TEntity? SelectedItem
    {
        get => _selectedItem;
        set
        {
            SetProperty(ref _selectedItem, value);
            RaiseCanExecuteChanged();
        }
    }
    
    public string SearchKeyword
    {
        get => _searchKeyword;
        set => SetProperty(ref _searchKeyword, value);
    }
    
    // 分页属性
    public int CurrentPage
    {
        get => _currentPage;
        set => SetProperty(ref _currentPage, value);
    }
    
    public int TotalPages => (int)Math.Ceiling((double)_totalCount / _pageSize);
    
    // 命令
    public DelegateCommand SearchCommand { get; }
    public DelegateCommand AddCommand { get; }
    public DelegateCommand<TEntity> EditCommand { get; }
    public DelegateCommand<TEntity> DeleteCommand { get; }
    public DelegateCommand<TEntity> ViewCommand { get; }
    public DelegateCommand FirstPageCommand { get; }
    public DelegateCommand PreviousPageCommand { get; }
    public DelegateCommand NextPageCommand { get; }
    public DelegateCommand LastPageCommand { get; }
    
    protected BaseManagementViewModel(TService service,
                                    IRegionManager regionManager,
                                    ICommonDialogService dialogService,
                                    IEventAggregator eventAggregator)
        : base(regionManager, dialogService, eventAggregator)
    {
        Service = service;
        InitializeCommands();
    }
    
    private void InitializeCommands()
    {
        SearchCommand = new DelegateCommand(ExecuteSearch);
        AddCommand = new DelegateCommand(ExecuteAdd);
        EditCommand = new DelegateCommand<TEntity>(ExecuteEdit, CanExecuteEdit);
        DeleteCommand = new DelegateCommand<TEntity>(ExecuteDelete, CanExecuteDelete);
        ViewCommand = new DelegateCommand<TEntity>(ExecuteView, CanExecuteView);
        
        // 分页命令
        FirstPageCommand = new DelegateCommand(FirstPage, () => CurrentPage > 1);
        PreviousPageCommand = new DelegateCommand(PreviousPage, () => CurrentPage > 1);
        NextPageCommand = new DelegateCommand(NextPage, () => CurrentPage < TotalPages);
        LastPageCommand = new DelegateCommand(LastPage, () => CurrentPage < TotalPages);
    }
    
    protected abstract Task LoadDataAsync();
    protected abstract void ExecuteAdd();
    protected abstract void ExecuteEdit(TEntity item);
    protected abstract void ExecuteDelete(TEntity item);
    protected virtual void ExecuteView(TEntity item) { }
    
    protected virtual bool CanExecuteEdit(TEntity item) => item != null;
    protected virtual bool CanExecuteDelete(TEntity item) => item != null;
    protected virtual bool CanExecuteView(TEntity item) => item != null;
}
```

### 4.2 ViewModels/LoginViewModel.cs

#### 登录视图模型
```csharp
public class LoginViewModel : BindableBase
{
    private readonly IAuthService _authService;
    private readonly ICommonDialogService _dialogService;
    
    private string _username = string.Empty;
    private string _password = string.Empty;
    private bool _rememberMe;
    private bool _isLoading;
    private string _errorMessage = string.Empty;
    
    public string Username
    {
        get => _username;
        set
        {
            SetProperty(ref _username, value);
            LoginCommand.RaiseCanExecuteChanged();
        }
    }
    
    public string Password
    {
        get => _password;
        set
        {
            SetProperty(ref _password, value);
            LoginCommand.RaiseCanExecuteChanged();
        }
    }
    
    public bool RememberMe
    {
        get => _rememberMe;
        set => SetProperty(ref _rememberMe, value);
    }
    
    public bool IsLoading
    {
        get => _isLoading;
        set => SetProperty(ref _isLoading, value);
    }
    
    public string ErrorMessage
    {
        get => _errorMessage;
        set
        {
            SetProperty(ref _errorMessage, value);
            RaisePropertyChanged(nameof(HasError));
        }
    }
    
    public bool HasError => !string.IsNullOrEmpty(ErrorMessage);
    
    public DelegateCommand LoginCommand { get; }
    
    public LoginViewModel(IAuthService authService, 
                         ICommonDialogService dialogService)
    {
        _authService = authService;
        _dialogService = dialogService;
        
        LoginCommand = new DelegateCommand(ExecuteLogin, CanExecuteLogin);
        
        // 加载记住的用户名
        LoadRememberedCredentials();
    }
    
    private async void ExecuteLogin()
    {
        try
        {
            IsLoading = true;
            ErrorMessage = string.Empty;
            
            var result = await _authService.LoginAsync(new LoginRequest
            {
                Username = Username,
                Password = Password,
                RememberMe = RememberMe
            });
            
            if (result.IsSuccess)
            {
                // 保存登录信息
```

```csharp
                await DialogService.ShowInformationAsync("挂号成功", "成功");
                await LoadDataAsync(); // 刷新数据
                
                // 清空选择
                _selectedPatient = null;
                _selectedDoctor = null;
            }
            else
            {
                await DialogService.ShowErrorAsync(result.ErrorMessage, "挂号失败");
            }
        }
        catch (Exception ex)
        {
            await DialogService.ShowErrorAsync($"挂号失败：{ex.Message}", "错误");
        }
        finally
        {
            IsLoading = false;
        }
    }
    
    private bool CanExecuteQuickRegistration()
    {
        return _selectedPatient != null && _selectedDoctor != null && !IsLoading;
    }
    
    /// <summary>患者签到</summary>
    private async void ExecuteCheckIn(RegistrationInfo registration)
    {
        if (registration == null || !registration.CanCheckIn) return;
        
        var confirm = await DialogService.ShowConfirmationAsync(
            $"确定为患者 {registration.PatientName} 签到吗？", "确认签到");
            
        if (confirm)
        {
            try
            {
                IsLoading = true;
                
                // 调用签到服务
                var result = await _registrationService.CheckInAsync(registration.Id);
                
                if (result.IsSuccess)
                {
                    await DialogService.ShowInformationAsync("签到成功", "成功");
                    await LoadDataAsync();
                }
                else
                {
                    await DialogService.ShowErrorAsync(result.ErrorMessage, "签到失败");
                }
            }
            catch (Exception ex)
            {
                await DialogService.ShowErrorAsync($"签到失败：{ex.Message}", "错误");
            }
            finally
            {
                IsLoading = false;
            }
        }
    }
    
    /// <summary>定时自动刷新</summary>
    private void StartAutoRefresh()
    {
        var timer = new System.Timers.Timer(30000); // 30秒刷新一次
        timer.Elapsed += async (s, e) => await LoadDataAsync();
        timer.Start();
    }

    {
        try
        {
            IsLoading = true;

            {

            await DialogService.ShowErrorAsync($"数据加载失败：{ex.Message}", "错误");
```

```csharp
    public PaymentManagementViewModel(IBillingService billingService,
                                    IPaymentService paymentService,
                                    IRegionManager regionManager,
                                    ICommonDialogService dialogService,
                                    IEventAggregator eventAggregator)
        : base(regionManager, dialogService, eventAggregator)
    {
        _billingService = billingService;
        _paymentService = paymentService;

        PaymentMethods = new ObservableCollection<PaymentMethodInfo>();
        
        InitializePaymentMethods();
        InitializeCommands();
        _ = LoadDataAsync();
    }
    
    private void InitializePaymentMethods()
    {
        PaymentMethods.Add(new PaymentMethodInfo { Name = "现金", Code = "CASH" });
        PaymentMethods.Add(new PaymentMethodInfo { Name = "支付宝", Code = "ALIPAY" });
        PaymentMethods.Add(new PaymentMethodInfo { Name = "微信", Code = "WECHAT" });
        PaymentMethods.Add(new PaymentMethodInfo { Name = "银行卡", Code = "CARD" });
        PaymentMethods.Add(new PaymentMethodInfo { Name = "医保", Code = "MEDICAL_INSURANCE" });
        
        SelectedPaymentMethod = PaymentMethods.First();
    }
    
    /// <summary>处理付款</summary>
    private async void ExecuteProcessPayment(BillInfo bill)
    {
        if (bill == null || SelectedPaymentMethod == null) return;
        
        // 验证付款金额
        if (SelectedPaymentMethod.Code == "CASH" && ReceivedAmount < bill.TotalAmount)
        {
            await DialogService.ShowWarningAsync("收款金额不足", "付款失败");
            return;
        }
        
        try
        {
            IsLoading = true;
            
            var paymentDto = new PaymentCreateDto
            {
                BillId = bill.Id,
                PaymentMethod = SelectedPaymentMethod.Code,
                Amount = bill.TotalAmount,
                ReceivedAmount = ReceivedAmount,
                ChangeAmount = ChangeAmount
            };
            
            var result = await _paymentService.ProcessPaymentAsync(paymentDto);
            
            if (result.IsSuccess)
            {
                await DialogService.ShowInformationAsync("付款成功", "成功");
                
                // 询问是否打印小票
                var printReceipt = await DialogService.ShowConfirmationAsync(
                    "付款成功，是否打印小票？", "打印确认");
                
                if (printReceipt)
                {
                    await PrintReceipt(result.Data);
                }
                
                await LoadDataAsync(); // 刷新数据
                ResetPaymentForm();
            }
            else
            {
                await DialogService.ShowErrorAsync(result.ErrorMessage, "付款失败");
            }
        }
        catch (Exception ex)
        {
            await DialogService.ShowErrorAsync($"付款失败：{ex.Message}", "错误");
        }
        finally
        {
            IsLoading = false;
        }
    }
    
    private void CalculateChange()
    {
        if (SelectedBill != null && SelectedPaymentMethod?.Code == "CASH")
        {
            ChangeAmount = ReceivedAmount - SelectedBill.TotalAmount;
        }
        else
        {
            ChangeAmount = 0;
        }
    }
    
    private async Task PrintReceipt(PaymentInfo payment)
    {
        try
        {
            var result = await _paymentService.PrintReceiptAsync(payment.Id);
            if (!result.IsSuccess)
            {
                await DialogService.ShowWarningAsync("小票打印失败", "提示");
            }
        }
        catch (Exception ex)
        {
            await DialogService.ShowErrorAsync($"打印小票失败：{ex.Message}", "错误");
        }
    }
```
