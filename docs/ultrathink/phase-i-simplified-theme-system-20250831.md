```csharp
// 简化的明暗主题切换逻辑
private async Task ExecuteToggleThemeAsync()
{
    var isDark = Application.Current.Resources.Contains("IsDarkTheme") && 
               (bool)Application.Current.Resources["IsDarkTheme"];
    
    if (isDark)
        ApplyLightTheme(); // 切换到浅色
    else
        ApplyDarkTheme();  // 切换到深色
}
```
