namespace LYBT.Models.Formula
{
    public class FormulaModel : BaseModel
    {
        public string Name { get; set; }  // 方名
        public string Category { get; set; }  // 分类：内科/外科/妇科/儿科
        public string Effect { get; set; }  // 功效
        public string Indication { get; set; }  // 主治
        public string Usage { get; set; }  // 用法用量
        public string CreatedBy { get; set; }  // 创建医生
        
        // 导航属性
        public virtual ICollection<FormulaItemModel> Items { get; set; }
    }
    
    public class FormulaItemModel : BaseModel
    {
        public Guid FormulaId { get; set; }
        public Guid HerbId { get; set; }
        public decimal Dosage { get; set; }  // 剂量
        public string Unit { get; set; }  // 单位
        public string? Note { get; set; }  // 备注（如：后下、先煎）
        
        // 导航属性
        public virtual FormulaModel Formula { get; set; }
        public virtual HerbModel Herb { get; set; }
    }
}

#### 2. 创建验方服务
```csharp
// 文件：src/Backend/Modules/LYBT.Module.Formula/Services/FormulaService.cs
public class FormulaService : IFormulaService
{
    public async Task<FormulaDto> CreateAsync(FormulaCreateDto dto)
    {
        var formula = new FormulaModel
        {
            Name = dto.Name,
            Category = dto.Category,
            Effect = dto.Effect,
            Indication = dto.Indication,
            Usage = dto.Usage ?? "水煎服，每日一剂",
            CreatedBy = _currentUser.Name,
            Items = dto.Items.Select(item => new FormulaItemModel
            {
                HerbId = item.HerbId,
                Dosage = item.Dosage,
                Unit = item.Unit,
                Note = item.Note
            }).ToList()
        };
        
        await _repository.AddAsync(formula);
        return _mapper.Map<FormulaDto>(formula);
    }
    
    public async Task<PrescriptionDto> ApplyFormulaToPrescription(Guid formulaId, Guid patientId)
    {
        var formula = await _repository.GetByIdAsync(formulaId);
        
        // 转换为处方
        var prescription = new PrescriptionModel
        {
            PatientId = patientId,
            Diagnosis = formula.Indication,
            Items = formula.Items.Select(item => new PrescriptionItemModel
            {
                HerbId = item.HerbId,
                Dosage = item.Dosage,
                Unit = item.Unit,
                Note = item.Note
            }).ToList()
        };
        
        return prescription;
    }
}
```

// 文件：src/Backend/Services/LYBT.WebAPI/Controllers/FormulaController.cs
[ApiController]
[Route("api/v1/[controller]")]
public class FormulaController : BaseController
{
    private readonly IFormulaService _formulaService;
    
    [HttpGet]
    public async Task<IActionResult> GetAll([FromQuery] string? category = null)
    {
        var formulas = await _formulaService.GetAllAsync(category);
        return Ok(formulas);
    }
    
    [HttpGet("{id}")]
    public async Task<IActionResult> GetById(Guid id)
    {
        var formula = await _formulaService.GetByIdAsync(id);
        return Ok(formula);
    }
    
    [HttpPost]
    public async Task<IActionResult> Create([FromBody] FormulaCreateDto dto)
    {
        var formula = await _formulaService.CreateAsync(dto);
        return Ok(formula);
    }
    
    [HttpPost("{id}/apply")]
    public async Task<IActionResult> ApplyToPrescription(Guid id, [FromBody] ApplyFormulaDto dto)
    {
        var prescription = await _formulaService.ApplyFormulaToPrescription(id, dto.PatientId);
        return Ok(prescription);
    }
}

// 文件：src/Frontend/Desktop/Modules/Consultation/ViewModels/QuickConsultationViewModel.cs
public class QuickConsultationViewModel : BindableBase
{
    private readonly IPatientService _patientService;
    private readonly IFormulaService _formulaService;
    private readonly IPrescriptionService _prescriptionService;
    private readonly IPrintService _printService;
    
    public ObservableCollection<PatientInfo> Patients { get; }
    public ObservableCollection<FormulaInfo> Formulas { get; }
    public ObservableCollection<PrescriptionItemInfo> PrescriptionItems { get; }
    
    public DelegateCommand SaveCommand { get; }
    public DelegateCommand PrintCommand { get; }
    public DelegateCommand ApplyFormulaCommand { get; }
    
    private async void ApplyFormula()
    {
        if (SelectedFormula == null || SelectedPatient == null) return;
        
        // 清空当前处方项
        PrescriptionItems.Clear();
        
        // 应用验方
        foreach (var item in SelectedFormula.Items)
        {
            PrescriptionItems.Add(new PrescriptionItemInfo
            {
                HerbId = item.HerbId,
                HerbName = item.HerbName,
                Dosage = item.Dosage,
                Unit = item.Unit,
                Note = item.Note
            });
        }
        
        // 设置诊断信息
        Diagnosis = SelectedFormula.Indication;
        Usage = SelectedFormula.Usage;
    }
}

### Day 5：处方打印

#### 1. 创建打印服务
```csharp
// 文件：src/Frontend/Desktop/Services/PrintService.cs
public class PrintService : IPrintService
{
    public void PrintPrescription(PrescriptionPrintModel model)
    {
        var document = CreatePrescriptionDocument(model);
        
        var printDialog = new PrintDialog();
        if (printDialog.ShowDialog() == true)
        {
            printDialog.PrintDocument(document.DocumentPaginator, "处方打印");
        }
    }
    
    private FlowDocument CreatePrescriptionDocument(PrescriptionPrintModel model)
    {
        var document = new FlowDocument();
        document.PageWidth = 210 * 96 / 25.4;  // A4宽度
        document.PageHeight = 297 * 96 / 25.4;  // A4高度
        
        // 标题
        var title = new Paragraph(new Run("凌隐宝堂中医诊所"))
        {
            FontSize = 20,
            FontWeight = FontWeights.Bold,
            TextAlignment = TextAlignment.Center
        };
        document.Blocks.Add(title);
        
        var subtitle = new Paragraph(new Run("中药处方笺"))
        {
            FontSize = 16,
            TextAlignment = TextAlignment.Center
        };
        document.Blocks.Add(subtitle);
        
        // 处方信息
        var info = new Paragraph();
        info.Inlines.Add($"处方号：{model.PrescriptionNo}\n");
        info.Inlines.Add($"日期：{model.Date:yyyy年MM月dd日}\n");
        info.Inlines.Add($"患者：{model.PatientName}  性别：{model.Gender}  年龄：{model.Age}\n");
        info.Inlines.Add($"诊断：{model.Diagnosis}\n");
        document.Blocks.Add(info);
        
        // 处方内容
        var rp = new Paragraph(new Run("Rp:")) { FontSize = 14 };
        document.Blocks.Add(rp);
        
        foreach (var item in model.Items)
        {
            var itemPara = new Paragraph();
            itemPara.Inlines.Add($"    {item.HerbName}  {item.Dosage}{item.Unit}");
            if (!string.IsNullOrEmpty(item.Note))
                itemPara.Inlines.Add($"（{item.Note}）");
            document.Blocks.Add(itemPara);
        }
        
        // 用法
        var usage = new Paragraph();
        usage.Inlines.Add($"\n共{model.Quantity}剂\n");
        usage.Inlines.Add($"用法：{model.Usage}\n");
        usage.Inlines.Add($"医嘱：{model.Advice}\n");
        document.Blocks.Add(usage);
        
        // 签名
        var signature = new Paragraph();
        signature.Inlines.Add($"\n\n医师：{model.DoctorName}");
        signature.TextAlignment = TextAlignment.Right;
        document.Blocks.Add(signature);
        
        return document;
    }
}
```
