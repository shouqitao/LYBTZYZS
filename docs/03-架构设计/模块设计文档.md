```csharp
public class AppDbContext : DbContext
{
    // 用户相关
    public DbSet<User> Users { get; set; }
    public DbSet<AdminSecret> AdminSecrets { get; set; }
    
    // 业务实体
    public DbSet<Patient> Patients { get; set; }
    public DbSet<Doctor> Doctors { get; set; }
    public DbSet<Registration> Registrations { get; set; }
    public DbSet<Prescription> Prescriptions { get; set; }
    public DbSet<PrescriptionDetail> PrescriptionDetails { get; set; }
    public DbSet<Herb> Herbs { get; set; }
    public DbSet<FormulaTemplate> FormulaTemplates { get; set; }
    
    protected override void OnModelCreating(ModelBuilder modelBuilder)
    {
        // 统一配置
        modelBuilder.ApplyConfigurationsFromAssembly(Assembly.GetExecutingAssembly());
        
        // 软删除全局过滤器
        foreach (var entityType in modelBuilder.Model.GetEntityTypes())
        {
            if (typeof(ISoftDelete).IsAssignableFrom(entityType.ClrType))
            {
                var parameter = Expression.Parameter(entityType.ClrType, "e");
                var property = Expression.Property(parameter, "IsDeleted");
                var filter = Expression.Lambda(
                    Expression.Equal(property, Expression.Constant(false)),
                    parameter);
                modelBuilder.Entity(entityType.ClrType).HasQueryFilter(filter);
            }
        }
    }
}
```

#### JWT认证服务
```csharp
public class JwtAuthenticationService : IJwtAuthenticationService
{
    public async Task<AuthResult> LoginAsync(LoginRequest request)
    {
        var user = await ValidateUserAsync(request.Username, request.Password);
        if (user == null)
            return AuthResult.Failure("用户名或密码错误");
            
        var token = GenerateJwtToken(user);
        var refreshToken = GenerateRefreshToken();
        
        await SaveRefreshTokenAsync(user.Id, refreshToken);
        
        return AuthResult.Success(new AuthData
        {
            Token = token,
            RefreshToken = refreshToken,
            ExpiresAt = DateTime.UtcNow.AddHours(8),
            User = user.ToDto()
        });
    }
    
    private string GenerateJwtToken(User user)
    {
        var claims = new[]
        {
            new Claim(ClaimTypes.NameIdentifier, user.Id.ToString()),
            new Claim(ClaimTypes.Name, user.UserName),
            new Claim(ClaimTypes.Role, user.Role),
            new Claim("RealName", user.RealName)
        };
        
        var key = new SymmetricSecurityKey(Encoding.UTF8.GetBytes(_jwtSettings.SecretKey));
        var creds = new SigningCredentials(key, SecurityAlgorithms.HmacSha256);
        
        var token = new JwtSecurityToken(
            issuer: _jwtSettings.Issuer,
            audience: _jwtSettings.Audience,
            claims: claims,
            expires: DateTime.UtcNow.AddHours(_jwtSettings.ExpiryHours),
            signingCredentials: creds);
            
        return new JwtSecurityTokenHandler().WriteToken(token);
    }
}
```

### 2.2 服务实现
```csharp
public class AuthService : IAuthService
{
    private readonly AppDbContext _context;
    private readonly IJwtAuthenticationService _jwtService;
    private readonly IPasswordHasher<User> _passwordHasher;
    
    public async Task<ServiceResult<AuthResponseDto>> LoginAsync(LoginDto dto)
    {
        try
        {
            // 1. 验证用户
            var user = await _context.Users
                .FirstOrDefaultAsync(u => u.UserName == dto.Username && u.IsActive);
                
            if (user == null)
                return ServiceResult<AuthResponseDto>.Failure("用户不存在或已禁用");
                
            // 2. 验证密码
            var result = _passwordHasher.VerifyHashedPassword(user, user.PasswordHash, dto.Password);
            if (result == PasswordVerificationResult.Failed)
                return ServiceResult<AuthResponseDto>.Failure("密码错误");
                
            // 3. 生成Token
            var authResult = await _jwtService.LoginAsync(new LoginRequest
            {
                Username = dto.Username,
                Password = dto.Password,
                RememberMe = dto.RememberMe
            });
            
            if (!authResult.IsSuccess)
                return ServiceResult<AuthResponseDto>.Failure(authResult.Message);
                
            // 4. 记录登录日志
            await LogLoginAsync(user.Id, dto.ClientInfo);
            
            return ServiceResult<AuthResponseDto>.Success(new AuthResponseDto
            {
                Token = authResult.Token,
                RefreshToken = authResult.RefreshToken,
                ExpiresAt = authResult.ExpiresAt,
                User = new UserInfoDto
                {
                    Id = user.Id,
                    UserName = user.UserName,
                    RealName = user.RealName,
                    Role = user.Role,
                    Email = user.Email,
                    PhoneNumber = user.PhoneNumber
                }
            });
        }
        catch (Exception ex)
        {
            return ServiceResult<AuthResponseDto>.Failure($"登录失败: {ex.Message}");
        }
    }
}
```

### 3.3 服务实现
```csharp
public class UserService : IUserService
{
    private readonly AppDbContext _context;
    private readonly IPasswordHasher<User> _passwordHasher;
    private readonly IMapper _mapper;
    
    public async Task<ServiceResult<UserDto>> CreateAsync(UserCreateDto dto)
    {
        try
        {
            // 检查用户名唯一性
            if (await _context.Users.AnyAsync(u => u.UserName == dto.UserName))
                return ServiceResult<UserDto>.Failure("用户名已存在");
                
            var user = new User
            {
                UserName = dto.UserName,
                RealName = dto.RealName,
                Email = dto.Email,
                PhoneNumber = dto.PhoneNumber,
                Role = dto.Role,
                CreateTime = DateTime.Now,
                CreatedBy = GetCurrentUserId()
            };
            
            // 生成默认密码
            var defaultPassword = GenerateDefaultPassword();
            user.PasswordHash = _passwordHasher.HashPassword(user, defaultPassword);
            
            _context.Users.Add(user);
            await _context.SaveChangesAsync();
            
            var result = _mapper.Map<UserDto>(user);
            result.DefaultPassword = defaultPassword; // 仅创建时返回
            
            return ServiceResult<UserDto>.Success(result);
        }
        catch (Exception ex)
        {
            return ServiceResult<UserDto>.Failure($"创建用户失败: {ex.Message}");
        }
    }
    
    private string GenerateDefaultPassword()
    {
        // 生成8位随机密码
        var chars = "ABCDEFGHJKMNPQRSTUVWXYZabcdefghijkmnpqrstuvwxyz23456789";
        var random = new Random();
        return new string(Enumerable.Repeat(chars, 8)
            .Select(s => s[random.Next(s.Length)]).ToArray());
    }
}
```

```csharp
public class PatientService : IPatientService
{
    public async Task<ServiceResult<PatientDto>> CreateAsync(PatientCreateDto dto)
    {
        try
        {
            // 生成患者编号
            var patientNo = await GeneratePatientNoAsync();
            
            var patient = new Patient
            {
                PatientNo = patientNo,
                Name = dto.Name,
                Gender = dto.Gender,
                BirthDate = dto.BirthDate,
                Age = CalculateAge(dto.BirthDate),
                IDNumber = dto.IDNumber,
                PhoneNumber = dto.PhoneNumber,
                Address = dto.Address,
                AllergyHistory = dto.AllergyHistory,
                PinYinCode = GeneratePinYinCode(dto.Name),
                CreateTime = DateTime.Now,
                CreatedBy = GetCurrentUserId()
            };
            
            _context.Patients.Add(patient);
            await _context.SaveChangesAsync();
            
            var result = _mapper.Map<PatientDto>(patient);
            return ServiceResult<PatientDto>.Success(result);
        }
        catch (Exception ex)
        {
            return ServiceResult<PatientDto>.Failure($"创建患者失败: {ex.Message}");
        }
    }
    
    private async Task<string> GeneratePatientNoAsync()
    {
        var today = DateTime.Now.ToString("yyyyMMdd");
        var count = await _context.Patients
            .CountAsync(p => p.PatientNo.StartsWith(today)) + 1;
        return $"{today}{count:D4}"; // 例：202401310001
    }
    
    private int? CalculateAge(DateTime? birthDate)
    {
        if (!birthDate.HasValue) return null;
        
        var today = DateTime.Today;
        var age = today.Year - birthDate.Value.Year;
        if (birthDate.Value.Date > today.AddYears(-age))
            age--;
        return age;
    }
    
    private string? GeneratePinYinCode(string name)
    {
        // 实现拼音码生成逻辑
        // 可以使用 PinYin.NET 等第三方库
        return name.ToUpper(); // 简化实现
    }
}
```

### 6.2 核心业务逻辑
```csharp
public class RegistrationService : IRegistrationService
{
    public async Task<ServiceResult<RegistrationDto>> CreateAsync(RegistrationCreateDto dto)
    {
        try
        {
            // 验证患者存在
            var patient = await _context.Patients.FindAsync(dto.PatientId);
            if (patient == null)
                return ServiceResult<RegistrationDto>.Failure("患者不存在");
                
            // 验证医生存在
            var doctor = await _context.Doctors.FindAsync(dto.DoctorId);
            if (doctor == null)
                return ServiceResult<RegistrationDto>.Failure("医生不存在");
                
            // 检查重复挂号
            var today = DateTime.Today;
            var existingReg = await _context.Registrations
                .AnyAsync(r => r.PatientId == dto.PatientId 
                          && r.DoctorId == dto.DoctorId 
                          && r.AppointmentDate.Date == today
                          && r.Status != RegistrationStatus.Cancelled);
                          
            if (existingReg)
                return ServiceResult<RegistrationDto>.Failure("患者今日已有该医生的挂号记录");
                
            // 生成挂号单号和排队号
            var regNumber = await GenerateRegistrationNumberAsync();
            var queueNumber = await GetNextQueueNumberAsync(dto.DoctorId, dto.AppointmentDate);
            
            var registration = new Registration
            {
                RegistrationNumber = regNumber,
                PatientId = dto.PatientId,
                DoctorId = dto.DoctorId,
                RegistrationType = dto.RegistrationType,
                RegistrationFee = doctor.RegistrationFee,
                Status = RegistrationStatus.Scheduled,
                AppointmentDate = dto.AppointmentDate,
                AppointmentTimeSlot = dto.AppointmentTimeSlot,
                QueueNumber = queueNumber,
                IsPaid = false,
                Remark = dto.Remark,
                CreateTime = DateTime.Now,
                CreatedBy = GetCurrentUserId()
            };
            
            _context.Registrations.Add(registration);
            await _context.SaveChangesAsync();
            
            var result = _mapper.Map<RegistrationDto>(registration);
            return ServiceResult<RegistrationDto>.Success(result);
        }
        catch (Exception ex)
        {
            return ServiceResult<RegistrationDto>.Failure($"挂号失败: {ex.Message}");
        }
    }
    
    private async Task<string> GenerateRegistrationNumberAsync()
    {
        var today = DateTime.Now.ToString("yyyyMMdd");
        var count = await _context.Registrations
            .CountAsync(r => r.RegistrationNumber.StartsWith($"R{today}")) + 1;
        return $"R{today}{count:D4}"; // 例：R202401310001
    }
    
    private async Task<int> GetNextQueueNumberAsync(Guid doctorId, DateTime appointmentDate)
    {
        var maxQueue = await _context.Registrations
            .Where(r => r.DoctorId == doctorId 
                   && r.AppointmentDate.Date == appointmentDate.Date)
            .MaxAsync(r => (int?)r.QueueNumber) ?? 0;
        return maxQueue + 1;
    }
}
```

```csharp
public class PrescriptionService : IPrescriptionService
{
    public async Task<ServiceResult<PrescriptionDto>> CreateAsync(PrescriptionCreateDto dto)
    {
        using var transaction = await _context.Database.BeginTransactionAsync();
        try
        {
            var prescriptionNo = await GeneratePrescriptionNoAsync();
            
            var prescription = new Prescription
            {
                PrescriptionNo = prescriptionNo,
                PatientId = dto.PatientId,
                DoctorId = dto.DoctorId,
                ConsultationId = dto.ConsultationId,
                Diagnosis = dto.Diagnosis,
                TCMDiagnosis = dto.TCMDiagnosis,
                Status = PrescriptionStatus.Draft,
                PrescribeTime = DateTime.Now,
                Usage = dto.Usage,
                Remark = dto.Remark,
                CreateTime = DateTime.Now,
                CreatedBy = GetCurrentUserId()
            };
            
            _context.Prescriptions.Add(prescription);
            await _context.SaveChangesAsync();
            
            // 添加处方明细
            decimal totalAmount = 0;
            foreach (var item in dto.Items)
            {
                var herb = await _context.Herbs.FindAsync(item.HerbId);
                if (herb == null) continue;
                
                var detail = new PrescriptionDetail
                {
                    PrescriptionId = prescription.Id,
                    HerbId = item.HerbId,
                    Quantity = item.Quantity,
                    Unit = item.Unit,
                    Price = herb.Price,
                    Subtotal = item.Quantity * herb.Price,
                    Usage = item.Usage,
                    CreateTime = DateTime.Now,
                    CreatedBy = GetCurrentUserId()
                };
                
                _context.PrescriptionDetails.Add(detail);
                totalAmount += detail.Subtotal;
            }
            
            prescription.TotalAmount = totalAmount;
            prescription.Status = PrescriptionStatus.Prescribed;
            
            await _context.SaveChangesAsync();
            await transaction.CommitAsync();
            
            var result = _mapper.Map<PrescriptionDto>(prescription);
            return ServiceResult<PrescriptionDto>.Success(result);
        }
        catch (Exception ex)
        {
            await transaction.RollbackAsync();
            return ServiceResult<PrescriptionDto>.Failure($"开方失败: {ex.Message}");
        }
    }
}
```
