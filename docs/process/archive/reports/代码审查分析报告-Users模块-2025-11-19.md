```csharp
// 每个ViewModel都重复此模式

{
    IsBusy = true;

}

{
    // 异常处理
}
finally
{
    IsBusy = false;
}
```

- 7个ViewModel中重复相同代码

- 应提取到基类或扩展方法

**影响**: 代码重复，维护成本高
**修复建议**:
```csharp
// UnifiedViewModelBase中添加
protected async Task ExecuteWithBusyAsync(Func<Task> action)
{
    try
    {
        IsBusy = true;
        await action();
    }
    catch (Exception ex)
    {
        HandleException(ex);
    }
    finally
    {
        IsBusy = false;
    }
}

// 使用
await ExecuteWithBusyAsync(async () =>
{
    await LoadUsersAsync();
});
```

```csharp
{
    if (SelectedUser == null || IsBusy) return false;

    // 只有管理员或用户本人可以编辑

    // 只能编辑活跃且未删除的用户
    var isEditable = SelectedUser.IsActive && !SelectedUser.IsDeleted;

    return hasPermission && isEditable;
}
```

```csharp
        return await response.Content.ReadAsAsync<UserDto>();
    }
    catch (HttpRequestException ex) when (ex.StatusCode == HttpStatusCode.NotFound)
    {
        return null; // 或抛出业务异常
    }
}
```
