### 2. 修改添加新验方模板的方法

**原代码：**
```csharp
private void AddNew()
{
    try
    {
        var herbService = _containerProvider.Resolve<IHerbService>();
        var formulaService = _containerProvider.Resolve<IFormulaTemplateService>();
        
        var dialog = new Views.AddFormulaTemplateDialog(herbService, formulaService);
        dialog.Owner = Application.Current.MainWindow;
        
        if (dialog.ShowDialog() == true)
        {
            RefreshCommand.Execute();
        }
    }
    catch (Exception ex)
    {
        MessageBox.Show($"打开新增验方模板对话框失败: {ex.Message}", "错误", 
            MessageBoxButton.OK, MessageBoxImage.Error);
    }
}
```

**重构后：**
```csharp
private async void AddNew()
{
    try
    {
        _dialogService.ShowDialog("AddFormulaTemplateDialog", result =>
        {
            if (result.Result == ButtonResult.OK)
            {
                RefreshCommand.Execute();
            }
        });
    }
    catch (Exception ex)
    {
        await _commonDialogService.ShowErrorAsync($"打开新增验方模板对话框失败: {ex.Message}");
    }
}
```

## 对话框实现 IDialogAware

### AddFormulaTemplateDialogViewModel

```csharp
public class AddFormulaTemplateDialogViewModel : BindableBase, IDialogAware
{
    private readonly IHerbService _herbService;
    private readonly IFormulaTemplateService _formulaTemplateService;
    private readonly ICommonDialogService _commonDialogService;

    public string Title => "新增验方模板";

    public event Action<IDialogResult> RequestClose;

    // 属性
    private string _formulaName;
    public string FormulaName
    {
        get => _formulaName;
        set => SetProperty(ref _formulaName, value);
    }

    // 命令
    public DelegateCommand SaveCommand { get; }
    public DelegateCommand CancelCommand { get; }

    public AddFormulaTemplateDialogViewModel(
        IHerbService herbService,
        IFormulaTemplateService formulaTemplateService,
        ICommonDialogService commonDialogService)
    {
        _herbService = herbService;
        _formulaTemplateService = formulaTemplateService;
        _commonDialogService = commonDialogService;

        SaveCommand = new DelegateCommand(ExecuteSave, CanExecuteSave)
            .ObservesProperty(() => FormulaName);
        CancelCommand = new DelegateCommand(ExecuteCancel);
    }

    public bool CanCloseDialog() => true;

    public void OnDialogClosed() { }

    public void OnDialogOpened(IDialogParameters parameters)
    {
        // 初始化数据
        LoadHerbs();
    }

    private async void ExecuteSave()
    {
        try
        {
            // 保存逻辑
            var result = await _formulaTemplateService.CreateAsync(/* ... */);
            
            if (result.IsSuccess)
            {
                RequestClose?.Invoke(new DialogResult(ButtonResult.OK));
            }
            else
            {
                await _commonDialogService.ShowErrorAsync($"保存失败：{result.ErrorMessage}");
            }
        }
        catch (Exception ex)
        {
            await _commonDialogService.ShowErrorAsync($"保存时发生错误：{ex.Message}");
        }
    }

    private void ExecuteCancel()
    {
        RequestClose?.Invoke(new DialogResult(ButtonResult.Cancel));
    }

    private bool CanExecuteSave()
    {
        return !string.IsNullOrWhiteSpace(FormulaName);
    }
}
```

## 重构步骤总结

1. **修改 ViewModel**
   - 注入 IDialogService 和 ICommonDialogService
   - 替换所有 MessageBox.Show 调用
   - 替换所有直接创建对话框的代码

2. **创建对话框 ViewModel**
   - 实现 IDialogAware 接口
   - 移动业务逻辑到 ViewModel
   - 使用数据绑定替代代码后置

3. **修改对话框 View**
   - 移除代码后置中的业务逻辑
   - 使用数据绑定
   - 简化为纯 UI 逻辑

4. **注册到容器**
   - 在模块中使用 RegisterDialog
   - 确保所有依赖服务已注册

5. **测试**
   - 验证对话框功能正常
   - 确保数据绑定工作正确
   - 测试错误处理
