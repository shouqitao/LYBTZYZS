   ```csharp
   public class ProjectAnalyzer
   {
       public ProjectInfo AnalyzeProject(string projectPath)
       {
           var content = File.ReadAllText(projectPath);
           var doc = XDocument.Parse(content);
   
           return new ProjectInfo
           {
               Path = projectPath,
               TargetFramework = GetTargetFramework(doc),
               PackageReferences = GetPackageReferences(doc),
               ProjectReferences = GetProjectReferences(doc),
               Properties = GetProperties(doc),
               IsSdkStyle = IsSdkStyleProject(doc)
           };
       }
   
       private List<PackageReference> GetPackageReferences(XDocument doc)
       {
           return doc.Descendants("PackageReference")
               .Select(element => new PackageReference
               {
                   Name = element.Attribute("Include")?.Value,
                   Version = element.Attribute("Version")?.Value,
                   IsVersionOverride = element.Attribute("VersionOverride") != null
               })
               .Where(pkg => !string.IsNullOrEmpty(pkg.Name))
               .ToList();
       }
   }
   ```

2. **依赖解析器** `DependencyResolver.cs`
   ```csharp
   public class DependencyResolver
   {
       public DependencyGraph BuildDependencyGraph(IEnumerable<ProjectInfo> projects)
       {
           var graph = new DependencyGraph();
   
           foreach (var project in projects)
           {
               var node = new DependencyNode(project);
               
               // 分析包依赖
               foreach (var package in project.PackageReferences)
               {
                   node.AddPackageDependency(package);
               }
   
               // 分析项目依赖
               foreach (var projectRef in project.ProjectReferences)
               {
                   node.AddProjectDependency(projectRef);
               }
   
               graph.AddNode(node);
           }
   
           return graph;
       }
   
       public List<VersionConflict> DetectVersionConflicts(DependencyGraph graph)
       {
           var conflicts = new List<VersionConflict>();
           var packageVersions = new Dictionary<string, List<(string Version, string Project)>>();
   
           // 收集所有包版本
           foreach (var node in graph.Nodes)
           {
               foreach (var package in node.PackageDependencies)
               {
                   if (!packageVersions.ContainsKey(package.Name))
                       packageVersions[package.Name] = new List<(string, string)>();
   
                   packageVersions[package.Name].Add((package.Version, node.Project.Path));
               }
           }
   
           // 检测版本冲突
           foreach (var packageGroup in packageVersions)
           {
               var uniqueVersions = packageGroup.Value
                   .Select(v => v.Version)
                   .Distinct()
                   .ToList();
   
               if (uniqueVersions.Count > 1)
               {
                   conflicts.Add(new VersionConflict
                   {
                       PackageName = packageGroup.Key,
                       ConflictingVersions = packageGroup.Value
                   });
               }
           }
   
           return conflicts;
       }
   }
   ```

3. **迁移执行器** `MigrationExecutor.cs`
   ```csharp
   public class MigrationExecutor
   {
       private readonly IBackupManager _backupManager;
       private readonly IStateManager _stateManager;
       private readonly ILogger _logger;
   
       public async Task<MigrationResult> ExecuteMigrationAsync(MigrationPlan plan)
       {
           var migrationId = Guid.NewGuid().ToString();
           
           try
           {
               // 1. 创建备份点
               _logger.LogInformation("创建迁移备份点: {MigrationId}", migrationId);
               var backupPoint = await _backupManager.CreateBackupAsync(migrationId, plan.ProjectPaths);
               
               // 2. 更新状态
               await _stateManager.SetMigrationStateAsync(migrationId, MigrationState.InProgress);
   
               // 3. 创建Directory.Packages.props
               await CreateDirectoryPackagesPropsAsync(plan);
   
               // 4. 批量转换项目文件
               var results = new List<ProjectMigrationResult>();
               foreach (var project in plan.Projects)
               {
                   var result = await MigrateProjectAsync(project, plan.PackageVersions);
                   results.Add(result);
   
                   if (!result.Success)
                   {
                       _logger.LogError("项目迁移失败: {ProjectPath}, 错误: {Error}", 
                           project.Path, result.Error);
                       
                       // 迁移失败，执行回滚
                       await _backupManager.RestoreBackupAsync(backupPoint.Id);
                       return new MigrationResult { Success = false, Error = result.Error };
                   }
               }
   
               // 5. 验证迁移结果
               var validationResult = await ValidateMigrationAsync(plan.Projects);
               if (!validationResult.Success)
               {
                   _logger.LogError("迁移验证失败: {Error}", validationResult.Error);
                   await _backupManager.RestoreBackupAsync(backupPoint.Id);
                   return validationResult;
               }
   
               // 6. 更新状态为完成
               await _stateManager.SetMigrationStateAsync(migrationId, MigrationState.Completed);
   
               return new MigrationResult 
               { 
                   Success = true, 
                   MigrationId = migrationId,
                   BackupPointId = backupPoint.Id,
                   ProjectResults = results
               };
           }
           catch (Exception ex)
           {
               _logger.LogError(ex, "迁移过程中发生异常");
               await _stateManager.SetMigrationStateAsync(migrationId, MigrationState.Failed);
               return new MigrationResult { Success = false, Error = ex.Message };
           }
       }
   }
   ```

#### Phase 2: 备份回滚系统开发 (0.5天)

1. **备份管理器** `BackupManager.cs`
   ```csharp
   public class BackupManager : IBackupManager
   {
       private readonly ILogger _logger;
       private readonly string _backupBasePath;
   
       public async Task<BackupPoint> CreateBackupAsync(string migrationId, IEnumerable<string> filePaths)
       {
           var backupId = $"backup-{migrationId}-{DateTime.Now:yyyyMMdd-HHmmss}";
           var backupPath = Path.Combine(_backupBasePath, backupId);
           
           Directory.CreateDirectory(backupPath);
   
           var backupPoint = new BackupPoint
           {
               Id = backupId,
               MigrationId = migrationId,
               CreatedAt = DateTime.UtcNow,
               BackupPath = backupPath,
               Files = new List<BackupFile>()
           };
   
           // 备份所有相关文件
           foreach (var filePath in filePaths)
           {
               if (File.Exists(filePath))
               {
                   var relativePath = Path.GetRelativePath(Environment.CurrentDirectory, filePath);
                   var backupFilePath = Path.Combine(backupPath, relativePath);
                   
                   Directory.CreateDirectory(Path.GetDirectoryName(backupFilePath));
                   File.Copy(filePath, backupFilePath, true);
   
                   backupPoint.Files.Add(new BackupFile
                   {
                       OriginalPath = filePath,
                       BackupPath = backupFilePath,
                       Hash = await CalculateFileHashAsync(filePath)
                   });
               }
           }
   
           // 保存备份元数据
           var metadataPath = Path.Combine(backupPath, "backup-metadata.json");
           var metadataJson = JsonSerializer.Serialize(backupPoint, new JsonSerializerOptions 
           { 
               WriteIndented = true 
           });
           
           await File.WriteAllTextAsync(metadataPath, metadataJson);
   
           _logger.LogInformation("创建备份完成: {BackupId}, 文件数量: {FileCount}", 
               backupId, backupPoint.Files.Count);
   
           return backupPoint;
       }
   
       public async Task<bool> RestoreBackupAsync(string backupId)
       {
           var backupPath = Path.Combine(_backupBasePath, backupId);
           var metadataPath = Path.Combine(backupPath, "backup-metadata.json");
   
           if (!File.Exists(metadataPath))
           {
               _logger.LogError("备份元数据不存在: {BackupId}", backupId);
               return false;
           }
   
           var metadataJson = await File.ReadAllTextAsync(metadataPath);
           var backupPoint = JsonSerializer.Deserialize<BackupPoint>(metadataJson);
   
           _logger.LogInformation("开始恢复备份: {BackupId}, 文件数量: {FileCount}", 
               backupId, backupPoint.Files.Count);
   
           // 恢复所有文件
           foreach (var file in backupPoint.Files)
           {
               try
               {
                   Directory.CreateDirectory(Path.GetDirectoryName(file.OriginalPath));
                   File.Copy(file.BackupPath, file.OriginalPath, true);
                   
                   // 验证文件完整性
                   var currentHash = await CalculateFileHashAsync(file.OriginalPath);
                   if (currentHash != file.Hash)
                   {
                       _logger.LogWarning("文件恢复后哈希值不匹配: {FilePath}", file.OriginalPath);
                   }
               }
               catch (Exception ex)
               {
                   _logger.LogError(ex, "恢复文件失败: {FilePath}", file.OriginalPath);
                   return false;
               }
           }
   
           _logger.LogInformation("备份恢复完成: {BackupId}", backupId);
           return true;
       }
   }
   ```

#### Phase 3: 脚本接口开发 (0.5天)

1. **主迁移脚本** `migrate-to-cpm.ps1`
   ```powershell
   [CmdletBinding()]
   param(
       [Parameter(Mandatory=$false)]
       [string]$ConfigPath = "config/migration-config.json",
       
       [Parameter(Mandatory=$false)]
       [switch]$DryRun = $false,
       
       [Parameter(Mandatory=$false)]
       [string]$SolutionPath = ".",
       
       [Parameter(Mandatory=$false)]
       [switch]$Force = $false
   )
   
   Write-Host "=== CPM自动化迁移工具 ===" -ForegroundColor Green
   
   try {
       # 1. 加载配置
       Write-Host "1. 加载迁移配置..." -ForegroundColor Yellow
       if (-not (Test-Path $ConfigPath)) {
           throw "迁移配置文件不存在: $ConfigPath"
       }
       
       $config = Get-Content $ConfigPath | ConvertFrom-Json
       
       # 2. 环境检查
       Write-Host "2. 环境检查..." -ForegroundColor Yellow
       $precheck = & ".\validate-migration.ps1" -Mode PreMigration -SolutionPath $SolutionPath
       if (-not $precheck.Success -and -not $Force) {
           Write-Error "环境检查失败，使用 -Force 参数强制执行"
           exit 1
       }
       
       # 3. 执行迁移
       Write-Host "3. 执行CPM迁移..." -ForegroundColor Yellow
       
       $migrationArgs = @{
           SolutionPath = $SolutionPath
           ConfigPath = $ConfigPath
           DryRun = $DryRun
       }
       
       $migrationTool = Join-Path $PSScriptRoot "../src/MigrationEngine/MigrationTool.exe"
       $result = & $migrationTool @migrationArgs
       
       if ($result.ExitCode -eq 0) {
           Write-Host "✅ CPM迁移成功完成" -ForegroundColor Green
           
           # 4. 迁移后验证
           Write-Host "4. 迁移后验证..." -ForegroundColor Yellow
           $validation = & ".\validate-migration.ps1" -Mode PostMigration -SolutionPath $SolutionPath
           
           if ($validation.Success) {
               Write-Host "✅ 迁移验证通过" -ForegroundColor Green
           } else {
               Write-Warning "⚠️ 迁移验证发现问题，建议检查日志"
           }
       } else {
           Write-Error "❌ CPM迁移失败，退出代码: $($result.ExitCode)"
           Write-Host "查看详细日志: logs/migration-$(Get-Date -Format 'yyyyMMdd').log" -ForegroundColor Cyan
           exit 1
       }
   }
   catch {
       Write-Error "迁移过程中发生异常: $($_.Exception.Message)"
       Write-Host "查看详细日志: logs/migration-$(Get-Date -Format 'yyyyMMdd').log" -ForegroundColor Cyan
       exit 1
   }
   ```

2. **回滚脚本** `rollback-cpm.ps1`
   ```powershell
   [CmdletBinding()]
   param(
       [Parameter(Mandatory=$true)]
       [string]$BackupId,
       
       [Parameter(Mandatory=$false)]
       [switch]$Force = $false,
       
       [Parameter(Mandatory=$false)]
       [string]$SolutionPath = "."
   )
   
   Write-Host "=== CPM迁移回滚工具 ===" -ForegroundColor Red
   
   try {
       # 1. 验证备份点
       Write-Host "1. 验证备份点..." -ForegroundColor Yellow
       
       $backupPath = "backup/$BackupId"
       if (-not (Test-Path $backupPath)) {
           throw "备份点不存在: $BackupId"
       }
       
       $metadataPath = Join-Path $backupPath "backup-metadata.json"
       if (-not (Test-Path $metadataPath)) {
           throw "备份元数据丢失: $BackupId"
       }
       
       $metadata = Get-Content $metadataPath | ConvertFrom-Json
       Write-Host "备份信息: 创建时间 $($metadata.CreatedAt), 文件数量 $($metadata.Files.Count)" -ForegroundColor Cyan
       
       # 2. 确认回滚
       if (-not $Force) {
           $confirm = Read-Host "确认回滚到备份点 '$BackupId'? 这将覆盖当前的项目文件 (y/N)"
           if ($confirm -ne 'y' -and $confirm -ne 'Y') {
               Write-Host "回滚操作已取消" -ForegroundColor Yellow
               exit 0
           }
       }
       
       # 3. 执行回滚
       Write-Host "3. 执行回滚操作..." -ForegroundColor Yellow
       
       $rollbackTool = Join-Path $PSScriptRoot "../src/BackupSystem/RollbackTool.exe"
       $result = & $rollbackTool -BackupId $BackupId -SolutionPath $SolutionPath
       
       if ($result.ExitCode -eq 0) {
           Write-Host "✅ 回滚操作成功完成" -ForegroundColor Green
           
           # 4. 验证回滚结果
           Write-Host "4. 验证回滚结果..." -ForegroundColor Yellow
           $validation = & ".\validate-migration.ps1" -Mode Rollback -SolutionPath $SolutionPath
           
           if ($validation.Success) {
               Write-Host "✅ 回滚验证通过" -ForegroundColor Green
           } else {
               Write-Warning "⚠️ 回滚验证发现问题，建议检查项目状态"
           }
       } else {
           Write-Error "❌ 回滚操作失败，退出代码: $($result.ExitCode)"
           exit 1
       }
   }
   catch {
       Write-Error "回滚过程中发生异常: $($_.Exception.Message)"
       exit 1
   }
   ```
