# DT-011 取消令牌支持 - 长操作取消机制

## 描述

为所有异步长时间操作添加CancellationToken支持，使用户能够取消正在进行的操作，改善用户体验并防止不必要的资源消耗。

## 验收标准

- [ ] 所有Service层异步方法支持CancellationToken参数
- [ ] ViewModel提供操作取消功能和UI反馈
- [ ] 长时间API调用可以被用户主动取消
- [ ] 取消操作不会导致数据不一致或系统异常
- [ ] 取消机制性能开销可接受

## 技术要求

### 实现范围

**Service层异步方法**：
- AuthService - 登录验证、权限检查等长操作
- UserService - 用户搜索、批量操作等
- PatientService - 患者搜索、数据导入导出等
- ConsultationService - 复杂诊断数据处理
- PrescriptionService - 处方生成、验证等
- HerbService - 药材搜索、批量更新等
- 所有涉及网络调用或数据库查询的异步方法

**ViewModel取消控制**：
- 为长操作添加"取消"按钮
- 操作进度指示和取消状态反馈
- 防止重复操作的UI状态管理

// Service层标准实现
public interface IUserService
{
    Task<ServiceResult<List<UserDto>>> SearchUsersAsync(
        UserSearchCriteria criteria, 
        CancellationToken cancellationToken = default);
        
    Task<ServiceResult<UserDto>> CreateUserAsync(
        CreateUserDto dto, 
        CancellationToken cancellationToken = default);
        
    Task<ServiceResult<bool>> BulkUpdateUsersAsync(
        List<UserUpdateDto> updates, 
        CancellationToken cancellationToken = default);
}

public class UserService : IUserService
{
    public async Task<ServiceResult<List<UserDto>>> SearchUsersAsync(
        UserSearchCriteria criteria, 
        CancellationToken cancellationToken = default)
    {
        try
        {
            cancellationToken.ThrowIfCancellationRequested();
            
            var users = await _repository.SearchAsync(criteria, cancellationToken);
            
            cancellationToken.ThrowIfCancellationRequested();
            
            var dtos = await _mapper.ProjectTo<UserDto>(users)
                .ToListAsync(cancellationToken);
                
            return ServiceResult<List<UserDto>>.Success(dtos);
        }
        catch (OperationCanceledException)
        {
            return ServiceResult<List<UserDto>>.Failure("操作已取消");
        }
    }
}

// ViewModel层取消控制
public class UserManagementViewModel : DisposableViewModelBase
{
    private CancellationTokenSource _currentOperationCts;
    
    public ICommand SearchUsersCommand { get; }
    public ICommand CancelOperationCommand { get; }
    
    public bool IsOperationInProgress { get; set; }
    public bool CanCancelOperation => IsOperationInProgress && _currentOperationCts != null;

    public async Task SearchUsersAsync()
    {
        if (IsOperationInProgress) return;

        _currentOperationCts?.Cancel();
        _currentOperationCts = new CancellationTokenSource();

        try
        {
            IsOperationInProgress = true;
            RaisePropertyChanged(nameof(CanCancelOperation));

            var result = await _userService.SearchUsersAsync(
                SearchCriteria, 
                _currentOperationCts.Token);

            if (result.Success)
            {
                Users = result.Data;
            }
            else if (!_currentOperationCts.Token.IsCancellationRequested)
            {
                // 只有非取消的错误才显示错误消息
                ShowErrorMessage(result.ErrorMessage);
            }
        }
        catch (OperationCanceledException)
        {
            ShowInfoMessage("搜索已取消");
        }
        finally
        {
            IsOperationInProgress = false;
            RaisePropertyChanged(nameof(CanCancelOperation));
            _currentOperationCts?.Dispose();
            _currentOperationCts = null;
        }
    }

    public void CancelCurrentOperation()
    {
        _currentOperationCts?.Cancel();
        ShowInfoMessage("正在取消操作...");
    }
}

// API Client层支持
public class ApiClientBase
{
    protected async Task<T> GetAsync<T>(string url, CancellationToken cancellationToken = default)
    {
        using var response = await _httpClient.GetAsync(url, cancellationToken);
        response.EnsureSuccessStatusCode();
        
        var json = await response.Content.ReadAsStringAsync(cancellationToken);
        return JsonSerializer.Deserialize<T>(json);
    }
}
