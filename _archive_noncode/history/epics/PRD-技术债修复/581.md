### 循环依赖检测

**依赖验证服务**：
```csharp
public static class DependencyValidator
{
    public static void ValidateModuleDependencies(IContainerRegistry containerRegistry)
    {
        var dependencies = ModuleDependencyMatrix.Dependencies;
        
        // 1. 检查循环依赖
        foreach (var module in dependencies.Keys)
        {
            if (HasCircularDependency(module, dependencies, new HashSet<string>()))
            {
                throw new InvalidOperationException($"模块 {module} 存在循环依赖");
            }
        }
        
        // 2. 验证依赖的模块已注册
        foreach (var (module, deps) in dependencies)
        {
            foreach (var dependency in deps)
            {
                if (!IsModuleRegistered(containerRegistry, dependency))
                {
                    throw new InvalidOperationException($"模块 {module} 依赖的模块 {dependency} 未注册");
                }
            }
        }
        
        Console.WriteLine("✅ 模块依赖关系验证通过");
    }
    
    private static bool HasCircularDependency(string module, Dictionary<string, string[]> dependencies, HashSet<string> visited)
    {
        if (visited.Contains(module))
        {
            return true; // 发现循环
        }
        
        visited.Add(module);
        
        if (dependencies.TryGetValue(module, out var deps))
        {
            foreach (var dep in deps)
            {
                if (HasCircularDependency(dep, dependencies, visited))
                {
                    return true;
                }
            }
        }
        
        visited.Remove(module);
        return false;
    }
}
```

public static class ServiceCollectionExtensions
{
    public static IContainerRegistry RegisterModulesInOrder(this IContainerRegistry containerRegistry)
    {
        try
        {
            // 按层次顺序注册，确保依赖关系正确
            Console.WriteLine("开始按依赖顺序注册模块...");
            
            // Layer 0 - 基础层
            Console.WriteLine("注册Layer 0 - 基础模块");
            Layer0Modules.RegisterLayer0(containerRegistry);
            
            // Layer 1 - 用户层  
            Console.WriteLine("注册Layer 1 - 用户模块");
            Layer1Modules.RegisterLayer1(containerRegistry);
            
            // Layer 2 - 业务层
            Console.WriteLine("注册Layer 2 - 业务模块");
            Layer2Modules.RegisterLayer2(containerRegistry);
            
            // Layer 3 - 诊疗层
            Console.WriteLine("注册Layer 3 - 诊疗模块");
            Layer3Modules.RegisterLayer3(containerRegistry);
            
            // 验证依赖关系
            Console.WriteLine("验证模块依赖关系...");
            DependencyValidator.ValidateModuleDependencies(containerRegistry);
            
            Console.WriteLine("✅ 所有模块注册完成");
            return containerRegistry;
        }
        catch (Exception ex)
        {
            Console.WriteLine($"❌ 模块注册失败: {ex.Message}");
            throw;
        }
    }
}
