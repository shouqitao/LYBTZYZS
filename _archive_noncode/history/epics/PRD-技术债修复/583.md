# DT-013 内存泄漏修复 - ViewModel内存泄漏防护

## 描述

修复ViewModel中的内存泄漏风险，特别是事件订阅未正确取消导致的内存泄漏。确保所有ViewModel正确实现IDisposable并管理资源生命周期。

## 验收标准

- [ ] 所有ViewModel都正确实现IDisposable接口
- [ ] 事件订阅在Dispose时正确取消
- [ ] 长期运行无内存泄漏现象
- [ ] 资源释放机制经过测试验证
- [ ] 添加内存泄漏检测和预防机制

## 技术要求

### 实现范围

**核心ViewModel修复**：
- AuthenticationViewModel - 认证相关事件订阅管理
- UserManagementViewModel - 用户管理事件和资源清理
- PatientManagementViewModel - 患者管理资源释放
- ConsultationViewModel - 诊疗流程事件管理
- PrescriptionViewModel - 处方编辑资源管理

```csharp
// ViewModel基类 - 标准化资源管理
public abstract class DisposableViewModelBase : BindableBase, IDisposable
{
    private readonly List<IDisposable> _disposables = new();
    private readonly List<SubscriptionToken> _eventSubscriptions = new();
    protected bool _disposed = false;

    // 安全的事件订阅管理
    protected SubscriptionToken SubscribeToEvent<T>(Action<T> handler)
    {
        var token = EventAggregator.GetEvent<PubSubEvent<T>>().Subscribe(handler);
        _eventSubscriptions.Add(token);
        return token;
    }

    // 统一资源管理
    protected void RegisterDisposable(IDisposable disposable)
    {
        if (disposable != null)
        {
            _disposables.Add(disposable);
        }
    }

    // 标准Dispose实现
    public virtual void Dispose()
    {
        if (_disposed) return;

        // 取消所有事件订阅
        foreach (var subscription in _eventSubscriptions)
        {
            subscription?.Dispose();
        }
        _eventSubscriptions.Clear();

        // 释放所有注册的资源
        foreach (var disposable in _disposables)
        {
            try
            {
                disposable?.Dispose();
            }
            catch (Exception ex)
            {
                // 日志记录但不抛出异常
                Debug.WriteLine($"资源释放异常: {ex.Message}");
            }
        }
        _disposables.Clear();

        _disposed = true;
        GC.SuppressFinalize(this);
    }
}

// ViewModel具体实现示例
public class UserManagementViewModel : DisposableViewModelBase
{
    private Timer _refreshTimer;
    private CancellationTokenSource _cancellationTokenSource;

    public UserManagementViewModel(IUserService userService, IEventAggregator eventAggregator)
    {
        // 安全的事件订阅
        SubscribeToEvent<UserUpdatedEvent>(OnUserUpdated);
        SubscribeToEvent<UserDeletedEvent>(OnUserDeleted);

        // 定时器资源管理
        _refreshTimer = new Timer(RefreshData, null, TimeSpan.Zero, TimeSpan.FromMinutes(5));
        RegisterDisposable(_refreshTimer);

        // 取消令牌管理
        _cancellationTokenSource = new CancellationTokenSource();
        RegisterDisposable(_cancellationTokenSource);
    }

    private void OnUserUpdated(UserUpdatedEvent userEvent)
    {
        if (_disposed) return; // 防止已释放对象的操作
        // 处理用户更新事件
    }

    // 异步操作的安全取消
    public async Task LoadUsersAsync()
    {
        if (_disposed) return;

        try
        {
            using var cts = CancellationTokenSource.CreateLinkedTokenSource(_cancellationTokenSource.Token);
            await _userService.GetUsersAsync(cts.Token);
        }
        catch (OperationCanceledException)
        {
            // 正常的取消操作，不需要处理
        }
    }
}
```
