## 技术要求

### 配置验证框架

**验证器接口设计**：
```csharp
public interface IConfigurationValidator
{
    string ConfigurationSection { get; }
    ValidationResult Validate(IConfiguration configuration);
}

public class ValidationResult
{
    public bool IsValid { get; set; }
    public List<ValidationError> Errors { get; set; } = new();
    public List<ValidationWarning> Warnings { get; set; } = new();
}

public class ValidationError
{
    public string ConfigPath { get; set; }
    public string Message { get; set; }
    public string SuggestedFix { get; set; }
}
```

**配置验证服务**：
```csharp
public class ConfigurationValidationService
{
    private readonly List<IConfigurationValidator> _validators;
    
    public ConfigurationValidationResult ValidateAll(IConfiguration configuration)
    {
        var results = new List<ValidationResult>();
        
        foreach (var validator in _validators)
        {
            try
            {
                var result = validator.Validate(configuration);
                results.Add(result);
            }
            catch (Exception ex)
            {
                // 验证器自身异常处理
            }
        }
        
        return new ConfigurationValidationResult(results);
    }
}
```

### 核心配置验证器

**数据库配置验证**：
```csharp
public class DatabaseConfigurationValidator : IConfigurationValidator
{
    public string ConfigurationSection => "ConnectionStrings";
    
    public ValidationResult Validate(IConfiguration configuration)
    {
        var result = new ValidationResult();
        
        // 验证连接字符串存在
        var connectionString = configuration.GetConnectionString("DefaultConnection");
        if (string.IsNullOrWhiteSpace(connectionString))
        {
            result.Errors.Add(new ValidationError
            {
                ConfigPath = "ConnectionStrings:DefaultConnection",
                Message = "数据库连接字符串未配置",
                SuggestedFix = "在appsettings.json中添加DefaultConnection配置"
            });
        }
        
        // 验证连接字符串格式
        if (!string.IsNullOrWhiteSpace(connectionString))
        {
            try
            {
                var builder = new SqlConnectionStringBuilder(connectionString);
                if (string.IsNullOrWhiteSpace(builder.DataSource))
                {
                    result.Errors.Add(new ValidationError
                    {
                        ConfigPath = "ConnectionStrings:DefaultConnection",
                        Message = "数据库服务器地址无效",
                        SuggestedFix = "检查Server参数设置"
                    });
                }
            }
            catch (Exception)
            {
                result.Errors.Add(new ValidationError
                {
                    ConfigPath = "ConnectionStrings:DefaultConnection", 
                    Message = "数据库连接字符串格式无效",
                    SuggestedFix = "检查连接字符串语法"
                });
            }
        }
        
        return result;
    }
}
```

**JWT配置验证**：
```csharp
public class JwtConfigurationValidator : IConfigurationValidator
{
    public string ConfigurationSection => "JwtSettings";
    
    public ValidationResult Validate(IConfiguration configuration)
    {
        var result = new ValidationResult();
        var jwtSection = configuration.GetSection("JwtSettings");
        
        // 验证SecretKey
        var secretKey = jwtSection["SecretKey"];
        if (string.IsNullOrWhiteSpace(secretKey))
        {
            result.Errors.Add(new ValidationError
            {
                ConfigPath = "JwtSettings:SecretKey",
                Message = "JWT密钥未配置",
                SuggestedFix = "设置至少32字符的安全密钥"
            });
        }
        else if (secretKey.Length < 32)
        {
            result.Errors.Add(new ValidationError
            {
                ConfigPath = "JwtSettings:SecretKey",
                Message = "JWT密钥长度不足，最少32字符",
                SuggestedFix = "使用更长的密钥提高安全性"
            });
        }
        
        // 验证Issuer
        var issuer = jwtSection["Issuer"];
        if (string.IsNullOrWhiteSpace(issuer))
        {
            result.Warnings.Add(new ValidationWarning
            {
                ConfigPath = "JwtSettings:Issuer",
                Message = "建议设置JWT颁发者信息"
            });
        }
        
        // 验证过期时间配置
        var expiryHours = jwtSection.GetValue<int>("ExpiryHours");
        if (expiryHours <= 0)
        {
            result.Errors.Add(new ValidationError
            {
                ConfigPath = "JwtSettings:ExpiryHours",
                Message = "JWT过期时间必须大于0",
                SuggestedFix = "设置合理的Token过期时间（推荐8小时）"
            });
        }
        
        return result;
    }
}
```

## 验证器覆盖范围

### 必需配置验证
- **数据库连接**：连接字符串格式、服务器可达性
- **JWT设置**：密钥长度、过期时间、颁发者
- **缓存配置**：内存限制、过期策略
- **日志配置**：文件路径、级别设置

### 条件依赖验证
- **HTTPS配置**：生产环境必须启用HTTPS
- **CORS设置**：开发环境允许跨域，生产环境限制
- **API版本**：确保版本配置一致性

### 安全配置验证
- **密钥强度**：检查各种密钥的安全强度
- **HTTPS重定向**：生产环境强制HTTPS
- **敏感信息检查**：避免配置文件中的明文密码

## 启动集成

### Program.cs 集成
```csharp
public static void Main(string[] args)
{
    var host = CreateHostBuilder(args).Build();
    
    // 启动时配置验证
    var configValidation = host.Services.GetRequiredService<ConfigurationValidationService>();
    var validationResult = configValidation.ValidateAll(host.Services.GetRequiredService<IConfiguration>());
    
    if (!validationResult.IsValid)
    {
        Console.WriteLine("配置验证失败，应用无法启动：");
        foreach (var error in validationResult.AllErrors)
        {
            Console.WriteLine($"错误: {error.ConfigPath} - {error.Message}");
            if (!string.IsNullOrEmpty(error.SuggestedFix))
            {
                Console.WriteLine($"建议: {error.SuggestedFix}");
            }
        }
        
        Environment.Exit(1);
        return;
    }
    
    if (validationResult.HasWarnings)
    {
        Console.WriteLine("配置警告：");
        foreach (var warning in validationResult.AllWarnings)
        {
            Console.WriteLine($"警告: {warning.ConfigPath} - {warning.Message}");
        }
    }
    
    host.Run();
}
```
