## 模板设计

### 主Service模板 (T4)
```csharp
<#@ template language="C#" #>
<#@ parameter name="ModuleName" type="System.String" #>
<#@ parameter name="QueryMethods" type="System.Collections.Generic.List<string>" #>
<#@ parameter name="BusinessMethods" type="System.Collections.Generic.List<string>" #>

using LYBT.Shared.Models.DTOs;

namespace LYBT.Desktop.Modules.<#= ModuleName #>.Services
{
    /// <summary>
    /// <#= ModuleName #> 模块服务 - UltraThink双层架构纯委托模式
    /// 自动生成于: <#= DateTime.Now.ToString("yyyy-MM-dd HH:mm:ss") #>
    /// </summary>
    [AutoRegister(ServiceLifetime.Scoped, ServiceTypes = new[] { typeof(I<#= ModuleName #>Service) })]
    public class <#= ModuleName #>Service : I<#= ModuleName #>Service
    {
        private readonly <#= ModuleName #>QueryService _queryService;
        private readonly <#= ModuleName #>BusinessService _businessService;
        private readonly ILogger<<#= ModuleName #>Service> _logger;

        public <#= ModuleName #>Service(
            <#= ModuleName #>QueryService queryService,
            <#= ModuleName #>BusinessService businessService,
            ILogger<<#= ModuleName #>Service> logger)
        {
            _queryService = queryService;
            _businessService = businessService;
            _logger = logger;
        }

        #region Query Layer Delegation
        
<# foreach(var method in QueryMethods) { #>
        public async Task<ServiceResult<PagedResult<<#= ModuleName #>Dto>>> <#= method #>Async(<#= GetMethodParameters(method) #>)
            => await _queryService.<#= method #>Async(<#= GetMethodArguments(method) #>);
            
<# } #>

        #endregion

        #region Business Layer Delegation
        
<# foreach(var method in BusinessMethods) { #>
        public async Task<ServiceResult<<#= GetReturnType(method) #>>> <#= method #>Async(<#= GetMethodParameters(method) #>)
            => await _businessService.<#= method #>Async(<#= GetMethodArguments(method) #>);
            
<# } #>

        #endregion
    }
}
```

## 脚手架工具

### PowerShell脚本
```powershell
# scripts/New-UltraThinkModule.ps1
param(
    [Parameter(Mandatory=$true)]
    [string]$ModuleName,
    
    [Parameter(Mandatory=$false)]
    [string[]]$Dependencies = @(),
    
    [Parameter(Mandatory=$false)]
    [switch]$IncludeTests = $true,
    
    [Parameter(Mandatory=$false)]
    [switch]$EnableAutoDiscovery = $true
)

function New-UltraThinkModule {
    Write-Host "Creating UltraThink module: $ModuleName" -ForegroundColor Green
    
    # 1. 验证模块名称合规性
    if (-not (Test-ModuleName $ModuleName)) {
        throw "Invalid module name: $ModuleName"
    }
    
    # 2. 创建模块目录结构
    New-ModuleDirectoryStructure -ModuleName $ModuleName
    
    # 3. 生成代码文件
    Invoke-TemplateGeneration -ModuleName $ModuleName -Dependencies $Dependencies
    
    # 4. 更新项目配置
    if ($EnableAutoDiscovery) {
        Update-ServiceRegistration -ModuleName $ModuleName
    }
    
    # 5. 生成测试文件
    if ($IncludeTests) {
        New-ModuleTests -ModuleName $ModuleName
    }
    
    # 6. 运行编译时验证
    Invoke-BuildTimeValidation
    
    Write-Host "Module $ModuleName created successfully!" -ForegroundColor Green
    Write-Host "Next steps:" -ForegroundColor Yellow
    Write-Host "  1. Implement business logic in ${ModuleName}BusinessService"
    Write-Host "  2. Add query methods in ${ModuleName}QueryService"  
    Write-Host "  3. Run tests: dotnet test"
}
```

[Test]
public async Task TemplateGenerator_ShouldGenerateValidUltraThinkModule()
{
    // Arrange
    var config = new ModuleConfiguration
    {
        ModuleName = "TestModule",
        EnableQueryLayer = true,
        EnableBusinessLayer = true,
        Dependencies = new[] { "IUserService", "IPatientService" }

    var generator = new ModuleTemplateGenerator();
