## 技术要求

### 阻塞性错误分类与修复策略

#### 类型1: 命名空间和类型错误 (CS0246, CS0103)

**问题特征**:
- CS0246: 找不到类型或命名空间名称
- CS0103: 名称在当前上下文中不存在

**修复策略**:

```csharp
// 常见问题1: 缺少using语句
// ❌ 错误示例
public class UserService
{
    private readonly IMapper _mapper; // CS0246: 找不到IMapper
}

// ✅ 修复方案
using AutoMapper;

public class UserService
{
    private readonly IMapper _mapper;
}

// 常见问题2: 命名空间变更后未更新引用
// ❌ 错误示例
using LYBT.Old.Namespace; // 命名空间已变更

// ✅ 修复方案
using LYBT.Shared.Models;
using LYBT.Infrastructure.Services;
```

#### 类型2: 成员访问错误 (CS1061)

**问题特征**:
- CS1061: 类型不包含定义且未找到可接受类型的扩展方法

**修复策略**:

```csharp
// 常见问题1: 方法名称变更后未同步
// ❌ 错误示例
await userService.GetUserByIdAsync(id); // 方法已重命名

// ✅ 修复方案
await userService.GetByIdAsync(id);

// 常见问题2: 接口方法签名不匹配
// ❌ 错误示例
public class UserService : IUserService
{
    // 缺少接口要求的方法实现
}

// ✅ 修复方案
public class UserService : IUserService
{
    public async Task<ServiceResult<User>> GetByIdAsync(Guid id)
    {
        // 完整实现接口方法
    }
}
```

#### 类型3: 项目引用错误 (MSB3644, MSB3246)

**问题特征**:
- MSB3644: 找不到引用程序集
- MSB3246: 已解析的文件具有错误的目标架构

**修复策略**:

```xml
<!-- 常见问题1: 项目引用路径错误 -->
<!-- ❌ 错误示例 -->
<ProjectReference Include="..\OldPath\LYBT.Entities.csproj" />

<!-- ✅ 修复方案 -->
<ProjectReference Include="..\..\Core\LYBT.Entities\LYBT.Entities.csproj" />

<!-- 常见问题2: 目标框架不匹配 -->
<!-- ❌ 错误示例 -->
<TargetFramework>net7.0</TargetFramework> <!-- 不匹配 -->

<!-- ✅ 修复方案 -->
<TargetFramework>net8.0</TargetFramework> <!-- 统一为net8.0 -->
```

#### 类型4: NuGet包引用错误 (NU1101, NU1605)

**问题特征**:
- NU1101: 无法找到包
- NU1605: 包降级警告(可能导致编译错误)

**修复策略**:

```xml
<!-- 常见问题1: 包版本冲突 -->
<!-- ❌ 错误示例 -->
<PackageReference Include="Microsoft.EntityFrameworkCore" Version="7.0.0" />
<PackageReference Include="Microsoft.EntityFrameworkCore.SqlServer" Version="8.0.17" />

<!-- ✅ 修复方案: 使用CCPM统一版本管理 -->

<!-- Directory.Packages.props中统一版本 -->
```

```csharp
// 修复脚本模板: fix-missing-namespaces.ps1
param(
    [string]$ProjectPath,
    [hashtable]$NamespaceMappings
)

# 常见命名空间映射
$commonMappings = @{
    'IMapper' = 'AutoMapper'
    'IMemoryCache' = 'Microsoft.Extensions.Caching.Memory'
    'ILogger' = 'Microsoft.Extensions.Logging'
    'IConfiguration' = 'Microsoft.Extensions.Configuration'
    'DbContext' = 'Microsoft.EntityFrameworkCore'
    'ServiceResult' = 'LYBT.Shared.Models'
    'ApiResponse' = 'LYBT.Shared.Models'
}

function Add-MissingUsings {
    param(
        [string]$FilePath,
        [string[]]$MissingTypes
    )
    
    $content = Get-Content $FilePath -Raw
    $usingsToAdd = @()
    
    foreach ($type in $MissingTypes) {
        if ($commonMappings.ContainsKey($type)) {
            $namespace = $commonMappings[$type]
            if ($content -notmatch "using $namespace;") {
                $usingsToAdd += "using $namespace;"
            }
        }
    }
    
    if ($usingsToAdd.Count -gt 0) {
        # 在现有using语句后添加新的using
        $newContent = $content -replace '(using .*?;\s*\n)', "$1`n$($usingsToAdd -join "`n")`n"
        Set-Content $FilePath $newContent -Encoding UTF8
        
        Write-Host "已为 $FilePath 添加 using 语句: $($usingsToAdd -join ', ')"
    }
}
```

```csharp
    }

        for _, error in cs0246_errors.iterrows():
            project_path = error['项目路径']
            error_desc = error['错误描述']
            
            # 提取类型名称
            type_match = re.search(r"name '(\w+)'", error_desc)
            if type_match:
                missing_type = type_match.group(1)
                
                if missing_type in type_mappings:
                    # 修复项目文件中的类型引用
                    fix_type_in_project(project_path, missing_type, type_mappings[missing_type])

def fix_type_in_project(project_path, old_type, new_type):
    """在项目中修复类型引用"""
    
    project_dir = os.path.dirname(project_path)
    
    # 遍历项目目录下的所有C#文件
    for root, dirs, files in os.walk(project_dir):
        for file in files:
            if file.endswith('.cs'):
                file_path = os.path.join(root, file)
                
                with open(file_path, 'r', encoding='utf-8') as f:
                    content = f.read()
                
                # 替换类型引用
                updated_content = re.sub(
                    rf'\b{old_type}\b',
                    new_type.split('.')[-1], # 只使用类名部分
                    content
                )
                
                if updated_content != content:
                    # 确保有正确的using语句
                    namespace = '.'.join(new_type.split('.')[:-1])
                    if f'using {namespace};' not in updated_content:
                        # 添加using语句
                        using_pattern = r'(using .*?;\s*\n)'
                        updated_content = re.sub(
                            using_pattern, 
                            f'\\1using {namespace};\n',
                            updated_content,
                            count=1
                        )
                    
                    with open(file_path, 'w', encoding='utf-8') as f:
                        f.write(updated_content)
                    
                    print(f"已修复 {file_path} 中的 {old_type} → {new_type}")

if __name__ == '__main__':
```

```csharp
// scripts/fix-method-signatures.py
import re
import os
import ast

def fix_method_signatures():
    """修复接口方法签名不匹配问题"""
    
    # 读取UltraThink架构接口定义
    interface_methods = {
        'IUserService': [
            'Task<ServiceResult<User>> GetByIdAsync(Guid id)',
            'Task<ServiceResult<List<User>>> GetAllAsync()',
            'Task<ServiceResult<User>> CreateAsync(UserCreateDto dto)',
            'Task<ServiceResult<User>> UpdateAsync(Guid id, UserUpdateDto dto)',
            'Task<ServiceResult<bool>> DeleteAsync(Guid id)'
        ],
        'IPatientService': [
            'Task<ServiceResult<Patient>> GetByIdAsync(Guid id)',
            'Task<ServiceResult<PagedResult<Patient>>> SearchAsync(PatientSearchDto criteria)',
            'Task<ServiceResult<Patient>> CreateAsync(PatientCreateDto dto)',
            'Task<ServiceResult<Patient>> UpdateAsync(Guid id, PatientUpdateDto dto)',
            'Task<ServiceResult<bool>> DisableAsync(Guid id)'
        ]
    }
    
    # 检查并修复实现类
    for interface_name, methods in interface_methods.items():
        service_name = interface_name.replace('I', '') # IUserService -> UserService
        fix_service_implementation(service_name, interface_name, methods)

def fix_service_implementation(service_name, interface_name, required_methods):
    """修复服务实现类的方法签名"""
    
    # 查找服务实现文件
    service_files = []
    for root, dirs, files in os.walk('.'):
        for file in files:
            if file.endswith(f'{service_name}.cs'):
                service_files.append(os.path.join(root, file))
    
    for service_file in service_files:
        print(f"检查服务实现: {service_file}")
        
        with open(service_file, 'r', encoding='utf-8') as f:
            content = f.read()
        
        # 检查类是否实现了接口
        if f': {interface_name}' in content:
            # 验证所有必需方法是否存在
            missing_methods = []
            for method in required_methods:
                method_name = extract_method_name(method)
                if method_name not in content:
                    missing_methods.append(method)
            
            if missing_methods:
                # 添加缺失的方法
                content = add_missing_methods(content, missing_methods)
                
                with open(service_file, 'w', encoding='utf-8') as f:
                    f.write(content)
                
                print(f"已为 {service_file} 添加缺失方法: {[extract_method_name(m) for m in missing_methods]}")

def extract_method_name(method_signature):
    """从方法签名中提取方法名"""
    match = re.search(r'(\w+)\(', method_signature)
    return match.group(1) if match else None

def add_missing_methods(class_content, missing_methods):
    """向类中添加缺失的方法"""
    
    # 找到类的结尾位置（最后一个}前）
    class_end_pattern = r'(\s*})(\s*)$'
    
    methods_code = '\n    // 自动生成的接口实现方法\n'
    
    for method in missing_methods:
        method_name = extract_method_name(method)
        if method_name:
            methods_code += f'''
    public async {method}
    {{
        // TODO: 实现 {method_name} 方法
        throw new NotImplementedException("方法 {method_name} 需要实现");
    }}
'''
    
    # 在类结尾前插入方法
    updated_content = re.sub(
        class_end_pattern,
        f'{methods_code}\\1\\2',
        class_content
    )
    
    return updated_content
```

```powershell
# scripts/fix-package-references.ps1
param(
    [string]$SolutionPath = "."
)

Write-Host "修复NuGet包引用问题..."

# CCPM标准包版本配置
$standardPackageVersions = @{
    'Microsoft.EntityFrameworkCore' = '8.0.17'
    'Microsoft.EntityFrameworkCore.SqlServer' = '8.0.17'
    'Microsoft.AspNetCore.Authentication.JwtBearer' = '8.0.17'
    'AutoMapper' = '15.0.1'
    'Prism.DryIoc' = '9.0.537'
    'Microsoft.Extensions.Caching.Memory' = '9.0.0'
}

# 获取所有项目文件
$projects = Get-ChildItem -Path $SolutionPath -Filter "*.csproj" -Recurse

foreach ($project in $projects) {
    Write-Host "处理项目: $($project.Name)"
    
    [xml]$projectXml = Get-Content $project.FullName
    $packageRefs = $projectXml.SelectNodes("//PackageReference")
    
    $updated = $false
    
    foreach ($packageRef in $packageRefs) {
        $packageName = $packageRef.Include
        $currentVersion = $packageRef.Version
        
        if ($standardPackageVersions.ContainsKey($packageName)) {
            $standardVersion = $standardPackageVersions[$packageName]
            
            if ($currentVersion -ne $standardVersion) {
                Write-Host "  更新包 $packageName: $currentVersion → $standardVersion"
                $packageRef.Version = $standardVersion
                $updated = $true
            }
        }
    }
    
    if ($updated) {
        $projectXml.Save($project.FullName)
        Write-Host "  已保存项目文件更新"
    }
}

Write-Host "NuGet包引用修复完成"
```

```csharp
using Microsoft.VisualStudio.TestTools.UnitTesting;
using System.Diagnostics;

[TestClass]
public class CompilationFixRegressionTests

    [TestMethod]
    public void VerifyAllProjectsCompileSuccessfully()

        {
            "LYBT.Server.sln",

            "LYBT.All.sln"

        {
            var result = CompileSolution(solution);
            Assert.IsTrue(result.Success, $"解决方案 {solution} 编译失败: {result.Output}");
        }
    }
    
    [TestMethod]
    public void VerifyNoBlockingErrorsRemain()
    {
        var blockingErrorCodes = new[] { "CS0246", "CS0103", "CS1061", "MSB3644", "MSB3246", "NU1101" };
        
        foreach (var errorCode in blockingErrorCodes)
        {
            var hasError = CheckForErrorInSolution(errorCode);
            Assert.IsFalse(hasError, $"仍然存在阻塞性错误: {errorCode}");
        }
    }
    
    private CompilationResult CompileSolution(string solutionPath)
    {
        var startInfo = new ProcessStartInfo
        {
            FileName = "dotnet",
            Arguments = $"build {solutionPath} --configuration Release --verbosity minimal",
            UseShellExecute = false,
            RedirectStandardOutput = true,
            RedirectStandardError = true
        };
        
        using var process = Process.Start(startInfo);
        process.WaitForExit();
        
        var output = process.StandardOutput.ReadToEnd();
        var error = process.StandardError.ReadToEnd();
        
        return new CompilationResult
        {
            Success = process.ExitCode == 0,
            Output = output + error
        };
    }
}

public class CompilationResult
{
    public bool Success { get; set; }
    public string Output { get; set; }
}
```
