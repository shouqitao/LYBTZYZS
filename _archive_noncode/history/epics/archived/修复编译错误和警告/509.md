#### 3.1 环境检查脚本

```python
# check_dev_environment.py
# 开发环境检查和诊断工具

import os
import sys
import subprocess
import json
from pathlib import Path
from typing import Dict, List, Tuple

class DevEnvironmentChecker:
    """开发环境检查器"""
    
    def __init__(self):
        self.results = []
        
    def check_dotnet_sdk(self) -> Tuple[bool, str]:
        """检查.NET SDK"""
        try:
            result = subprocess.run(['dotnet', '--version'], 
                                  capture_output=True, text=True)
            if result.returncode == 0:
                version = result.stdout.strip()
                if version.startswith('8.'):
                    return True, f"✅ .NET SDK {version} - 正常"
                else:
                    return False, f"❌ .NET SDK版本不匹配: {version} (需要8.x)"
            else:
                return False, f"❌ .NET SDK未安装或无法访问"
        except FileNotFoundError:
            return False, "❌ .NET SDK未安装"
    
    def check_visual_studio(self) -> Tuple[bool, str]:
        """检查Visual Studio"""
        vs_paths = [
            r"C:\Program Files\Microsoft Visual Studio\2022\Enterprise",
            r"C:\Program Files\Microsoft Visual Studio\2022\Professional",
            r"C:\Program Files\Microsoft Visual Studio\2022\Community"
        ]
        
        for vs_path in vs_paths:
            if os.path.exists(vs_path):
                devenv_path = os.path.join(vs_path, "Common7", "IDE", "devenv.exe")
                if os.path.exists(devenv_path):
                    edition = vs_path.split('\\')[-1]
                    return True, f"✅ Visual Studio 2022 {edition} - 已安装"
        
        return False, "❌ Visual Studio 2022未找到"
    
    def check_sql_server(self) -> Tuple[bool, str]:
        """检查SQL Server"""
        try:
            # 检查SQL Server服务
            result = subprocess.run([
                'sc', 'query', 'MSSQLSERVER'
            ], capture_output=True, text=True, shell=True)
            
            if result.returncode == 0 and 'RUNNING' in result.stdout:
                return True, "✅ SQL Server服务 - 运行中"
            else:
                return False, "❌ SQL Server服务未运行"
        except:
            return False, "❌ 无法检查SQL Server状态"
    
    def check_git_config(self) -> Tuple[bool, str]:
        """检查Git配置"""
        try:
            # 检查用户配置
            name_result = subprocess.run(['git', 'config', '--global', 'user.name'], 
                                       capture_output=True, text=True)
            email_result = subprocess.run(['git', 'config', '--global', 'user.email'], 
                                        capture_output=True, text=True)
            
            if name_result.returncode == 0 and email_result.returncode == 0:
                name = name_result.stdout.strip()
                email = email_result.stdout.strip()
                if name and email:
                    return True, f"✅ Git配置 - {name} <{email}>"
                else:
                    return False, "❌ Git用户信息未配置完整"
            else:
                return False, "❌ Git配置检查失败"
        except FileNotFoundError:
            return False, "❌ Git未安装"
    
    def check_node_npm(self) -> Tuple[bool, str]:
        """检查Node.js和npm（如果需要前端工具）"""
        try:
            node_result = subprocess.run(['node', '--version'], 
                                       capture_output=True, text=True)
            npm_result = subprocess.run(['npm', '--version'], 
                                      capture_output=True, text=True)
            
            if node_result.returncode == 0 and npm_result.returncode == 0:
                node_version = node_result.stdout.strip()
                npm_version = npm_result.stdout.strip()
                return True, f"✅ Node.js {node_version}, npm {npm_version}"
            else:
                return False, "❌ Node.js/npm未正确安装"
        except FileNotFoundError:
            return False, "⚠️  Node.js/npm未安装（可选）"
    
    def check_project_structure(self) -> Tuple[bool, str]:
        """检查项目结构"""
        required_paths = [
            "src/Server/Services/LYBT.WebAPI/LYBT.WebAPI.csproj",
            "src/Client/Desktop/LYBT.Desktop.csproj",
            "src/Server/Core/LYBT.Infrastructure/LYBT.Infrastructure.csproj",
            "LYBT.Server.sln",
            "LYBT.Desktop.sln"
        ]
        
        missing_paths = []
        for path in required_paths:
            if not os.path.exists(path):
                missing_paths.append(path)
        
        if not missing_paths:
            return True, "✅ 项目结构 - 完整"
        else:
            return False, f"❌ 缺少文件: {', '.join(missing_paths)}"
    
    def check_environment_variables(self) -> Tuple[bool, str]:
        """检查环境变量"""
        important_vars = {
            'PATH': lambda x: 'dotnet' in x.lower(),
            'ASPNETCORE_ENVIRONMENT': lambda x: True,  # 可选
            'ConnectionStrings__DefaultConnection': lambda x: True  # 可选
        }
        
        issues = []
        for var, validator in important_vars.items():
            value = os.environ.get(var)
            if var == 'PATH' and not validator(value or ''):
                issues.append(f"{var} 中未包含 dotnet")
        
        if not issues:
            return True, "✅ 环境变量 - 正常"
        else:
            return False, f"❌ 环境变量问题: {', '.join(issues)}"
    
    def run_all_checks(self) -> List[str]:
        """运行所有检查"""
        checks = [
            (".NET SDK", self.check_dotnet_sdk),
            ("Visual Studio", self.check_visual_studio),
            ("SQL Server", self.check_sql_server),
            ("Git配置", self.check_git_config),
            ("Node.js/npm", self.check_node_npm),
            ("项目结构", self.check_project_structure),
            ("环境变量", self.check_environment_variables)
        ]
        
        results = []
        print("🔍 开发环境检查中...")
        print("=" * 50)
        
        for name, check_func in checks:
            success, message = check_func()
            results.append(message)
            print(f"{name}: {message}")
        
        print("=" * 50)
        
        # 统计结果
        success_count = sum(1 for r in results if r.startswith("✅"))
        warning_count = sum(1 for r in results if r.startswith("⚠️"))
        error_count = sum(1 for r in results if r.startswith("❌"))
        
        print(f"📊 检查完成: ✅{success_count} ⚠️{warning_count} ❌{error_count}")
        
        if error_count == 0:
            print("🎉 开发环境配置良好！")
        elif error_count <= 2:
            print("⚠️  开发环境基本正常，建议修复标记的问题")
        else:
            print("❌ 开发环境存在重要问题，请修复后再继续开发")
        
        return results

if __name__ == "__main__":
    checker = DevEnvironmentChecker()
    checker.run_all_checks()
```

#### 3.2 快速启动脚本

```powershell
# quick-start.ps1
# LYBT项目快速启动脚本

param(
    [Parameter(Mandatory=$false)]
    [ValidateSet("dev", "test", "prod")]
    [string]$Environment = "dev",
    
    [Parameter(Mandatory=$false)]
    [switch]$SkipBuild,
    
    [Parameter(Mandatory=$false)]
    [switch]$OpenVS,
    
    [Parameter(Mandatory=$false)]
    [switch]$StartAPI,
    
    [Parameter(Mandatory=$false)]
    [switch]$StartDesktop,
    
    [Parameter(Mandatory=$false)]
    [switch]$All
)

# 设置错误处理
$ErrorActionPreference = "Stop"

# 全局变量
$ProjectRoot = $PSScriptRoot
$WebAPIProject = "src\Server\Services\LYBT.WebAPI\LYBT.WebAPI.csproj"
$DesktopProject = "src\Client\Desktop\LYBT.Desktop.csproj"

function Write-StepInfo($message) {
    Write-Host "🚀 $message" -ForegroundColor Cyan
}

function Write-Success($message) {
    Write-Host "✅ $message" -ForegroundColor Green
}

function Write-Warning($message) {
    Write-Host "⚠️  $message" -ForegroundColor Yellow
}

function Write-Error($message) {
    Write-Host "❌ $message" -ForegroundColor Red
}

function Test-Prerequisites {
    Write-StepInfo "检查先决条件..."
    
    # 检查.NET SDK
    try {
        $dotnetVersion = dotnet --version
        if ($dotnetVersion -notmatch "^8\.") {
            throw ".NET 8 SDK is required, found: $dotnetVersion"
        }
        Write-Success ".NET SDK $dotnetVersion"
    }
    catch {
        Write-Error "❌ .NET 8 SDK未找到或版本不匹配"
        exit 1
    }
    
    # 检查项目文件
    if (!(Test-Path $WebAPIProject)) {
        Write-Error "WebAPI项目文件未找到: $WebAPIProject"
        exit 1
    }
    
    if (!(Test-Path $DesktopProject)) {
        Write-Error "Desktop项目文件未找到: $DesktopProject"
        exit 1
    }
    
    Write-Success "先决条件检查通过"
}

function Build-Projects {
    if ($SkipBuild) {
        Write-Warning "跳过构建步骤"
        return
    }
    
    Write-StepInfo "构建项目..."
    
    # 构建服务端
    Write-Host "构建服务端..." -ForegroundColor Yellow
    dotnet build "LYBT.Server.sln" --configuration Release --verbosity quiet
    if ($LASTEXITCODE -ne 0) {
        Write-Error "服务端构建失败"
        exit 1
    }
    
    # 构建客户端
    Write-Host "构建客户端..." -ForegroundColor Yellow
    dotnet build "LYBT.Desktop.sln" --configuration Release --verbosity quiet
    if ($LASTEXITCODE -ne 0) {
        Write-Error "客户端构建失败"
        exit 1
    }
    
    Write-Success "项目构建完成"
}

function Set-Environment($env) {
    Write-StepInfo "设置环境变量为: $env"
    
    switch ($env) {
        "dev" {
            $env:ASPNETCORE_ENVIRONMENT = "Development"
            $env:ASPNETCORE_URLS = "https://localhost:7001;http://localhost:5001"
        }
        "test" {
            $env:ASPNETCORE_ENVIRONMENT = "Testing"
            $env:ASPNETCORE_URLS = "https://localhost:7002;http://localhost:5002"
        }
        "prod" {
            $env:ASPNETCORE_ENVIRONMENT = "Production"
            $env:ASPNETCORE_URLS = "https://localhost:7000;http://localhost:5000"
        }
    }
    
    Write-Success "环境设置: $($env:ASPNETCORE_ENVIRONMENT)"
}

function Start-WebAPI {
    Write-StepInfo "启动Web API..."
    
    $apiProcess = Start-Process -FilePath "dotnet" -ArgumentList @(
        "run",
        "--project", $WebAPIProject,
        "--configuration", "Release"
    ) -NoNewWindow -PassThru
    
    Write-Success "Web API已启动 (PID: $($apiProcess.Id))"
    Write-Host "API地址: $($env:ASPNETCORE_URLS)" -ForegroundColor Cyan
    Write-Host "Swagger: https://localhost:7001/swagger" -ForegroundColor Cyan
    
    return $apiProcess
}

function Start-DesktopClient {
    Write-StepInfo "启动桌面客户端..."
    
    $desktopProcess = Start-Process -FilePath "dotnet" -ArgumentList @(
        "run",
        "--project", $DesktopProject,
        "--configuration", "Release"
    ) -NoNewWindow -PassThru
    
    Write-Success "桌面客户端已启动 (PID: $($desktopProcess.Id))"
    
    return $desktopProcess
}

function Open-VisualStudio {
    Write-StepInfo "启动Visual Studio..."
    
    $vsPath = @(
        "${env:ProgramFiles}\Microsoft Visual Studio\2022\Enterprise\Common7\IDE\devenv.exe",
        "${env:ProgramFiles}\Microsoft Visual Studio\2022\Professional\Common7\IDE\devenv.exe",
        "${env:ProgramFiles}\Microsoft Visual Studio\2022\Community\Common7\IDE\devenv.exe"
    ) | Where-Object { Test-Path $_ } | Select-Object -First 1
    
    if ($vsPath) {
        Start-Process -FilePath $vsPath -ArgumentList "LYBT.Server.sln"
        Write-Success "Visual Studio已启动"
    }
    else {
        Write-Warning "未找到Visual Studio 2022"
    }
}

# 主执行流程
function Main {
    Write-Host @"
🏥 LYBT诊所管理系统 - 快速启动器
========================================
环境: $Environment
构建: $(if($SkipBuild) {"跳过"} else {"执行"})
目标: $(if($All) {"全部"} elseif($StartAPI -and $StartDesktop) {"API + Desktop"} elseif($StartAPI) {"API"} elseif($StartDesktop) {"Desktop"} elseif($OpenVS) {"Visual Studio"} else {"默认"})
========================================
"@ -ForegroundColor Green
    
    # 检查先决条件
    Test-Prerequisites
    
    # 设置环境
    Set-Environment -env $Environment
    
    # 构建项目
    Build-Projects
    
    $processes = @()
    
    # 根据参数决定启动什么
    if ($All -or $StartAPI -or (!$StartDesktop -and !$OpenVS)) {
        $processes += Start-WebAPI
        Start-Sleep -Seconds 3  # 等待API启动
    }
    
    if ($All -or $StartDesktop) {
        $processes += Start-DesktopClient
    }
    
    if ($All -or $OpenVS) {
        Open-VisualStudio
    }
    
    # 如果有进程在运行，等待用户输入后清理
    if ($processes.Count -gt 0) {
        Write-Host "`n按任意键停止所有进程..." -ForegroundColor Yellow
        $null = $Host.UI.RawUI.ReadKey("NoEcho,IncludeKeyDown")
        
        Write-StepInfo "正在停止进程..."
        foreach ($process in $processes) {
            if (!$process.HasExited) {
                $process.Kill()
                Write-Success "已停止进程 (PID: $($process.Id))"
            }
        }
    }
    
    Write-Success "启动器执行完成！"
}

# 执行主函数
try {
    Main
}
catch {
    Write-Error "启动器执行失败: $($_.Exception.Message)"
    exit 1
}
```

```python
# generate_entity.py
# 智能实体类生成器

import os
import re
from datetime import datetime
from typing import Dict, List, Optional

class EntityGenerator:
    """实体生成器"""
    
    def __init__(self, namespace: str = "LYBT.Entities"):
        self.namespace = namespace
        self.template_base = """using System.ComponentModel.DataAnnotations;
using System.ComponentModel.DataAnnotations.Schema;

namespace {namespace};

/// <summary>
/// {description}
/// </summary>
[Table("{table_name}")]
public class {class_name}
{{
{properties}
}}"""
        
        self.property_template = "    /// <summary>\n    /// {description}\n    /// </summary>\n{attributes}    public {type} {name} {{ get; set; }}"
        
    def generate_property(self, name: str, type_name: str, 
                         description: str = "", 
                         is_required: bool = False,
                         max_length: Optional[int] = None,
                         is_key: bool = False) -> str:
        """生成属性代码"""
        attributes = []
        
        if is_key:
            attributes.append("    [Key]")
            
        if is_required:
            attributes.append("    [Required]")
            
        if max_length:
            attributes.append(f"    [MaxLength({max_length})]")
            
        attributes_str = "\n".join(attributes)
        if attributes_str:
            attributes_str += "\n"
            
        return self.property_template.format(
            description=description or f"{name}属性",
            attributes=attributes_str,
            type=type_name,
            name=name
        )
    
    def generate_entity(self, class_name: str, 
                       properties: List[Dict],
                       description: str = "",
                       table_name: Optional[str] = None) -> str:
        """生成完整实体类"""
        if not table_name:
            # 将PascalCase转为snake_case
            table_name = re.sub('([a-z0-9])([A-Z])', r'\1_\2', class_name).lower()
        
        property_codes = []
        for prop in properties:
            prop_code = self.generate_property(
                name=prop['name'],
                type_name=prop['type'],
                description=prop.get('description', ''),
                is_required=prop.get('required', False),
                max_length=prop.get('max_length'),
                is_key=prop.get('is_key', False)
            )
            property_codes.append(prop_code)
        
        return self.template_base.format(
            namespace=self.namespace,
            description=description or f"{class_name}实体",
            table_name=table_name,
            class_name=class_name,
            properties="\n\n".join(property_codes)
        )

def generate_lybt_entities():
    """生成LYBT系统标准实体"""
    generator = EntityGenerator("LYBT.Entities")
    
    # 示例：生成预约实体

            'required': True
        },

            'required': True
        },

            'name': 'Status',
```
