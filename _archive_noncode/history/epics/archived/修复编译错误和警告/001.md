### 问题分类标准

#### 编译错误分类 (阻塞性)

1. **CS类型错误**:
   - CS0246: 找不到类型或命名空间名称
   - CS0103: 名称在当前上下文中不存在
   - CS1061: 未包含定义或扩展方法

2. **项目引用错误**:
   - MSB3644: 找不到引用程序集
   - MSB3246: 已解析的文件具有错误的目标架构

3. **包引用错误**:
   - NU1101: 无法找到包
   - NU1605: 包降级警告

#### 编译警告分类 (非阻塞性)

1. **可空性警告 (CS8xxx系列)**:
   - CS8600-CS8999: 可空引用类型相关警告

2. **代码质量警告**:
   - CS1998: 异步方法缺少await操作符
   - CS0649: 字段从未赋值
   - CS0168: 声明了变量但从未使用

3. **过时API警告**:
   - CS0618: 类或成员已过时
   - CS0612: 类或成员已过时

## 实施步骤

### Phase 1: 扫描脚本开发 (2小时)

**1.1 创建MSBuild扫描脚本**

```powershell
# scripts/scan-compilation-status.ps1
Param(
    [string]$SolutionPath = ".",
    [string]$OutputFormat = "json",
    [switch]$DetailedMode
)

# 扫描所有.csproj文件
$projects = Get-ChildItem -Path $SolutionPath -Filter "*.csproj" -Recurse

foreach ($project in $projects) {
    Write-Host "扫描项目: $($project.FullName)"
    
    # 执行MSBuild编译并捕获输出
    $buildResult = dotnet build $project.FullName --verbosity detailed 2>&1
    
    # 解析编译错误和警告
    $errors = $buildResult | Select-String "error CS\d+"
    $warnings = $buildResult | Select-String "warning CS\d+"
    
    # 记录结果
    $result = @{
        Project = $project.Name
        Path = $project.FullName
        Errors = $errors.Count
        Warnings = $warnings.Count
        Details = @{
            Errors = $errors | ForEach-Object { $_.ToString() }
            Warnings = $warnings | ForEach-Object { $_.ToString() }
        }
    }
    
    # 输出结果
    $result | ConvertTo-Json -Depth 3 | Out-File -FilePath "compilation-scan-$($project.BaseName).json" -Encoding UTF8
}
```

**1.2 创建汇总分析脚本**

```python
# scripts/analyze-compilation-results.py
import json
import os
from collections import defaultdict
import pandas as pd

def analyze_compilation_results():
    """分析编译扫描结果，生成汇总报告"""
    
    results = []
    error_summary = defaultdict(int)
    warning_summary = defaultdict(int)
    
    # 读取所有扫描结果文件
    for filename in os.listdir('.'):
        if filename.startswith('compilation-scan-') and filename.endswith('.json'):
            with open(filename, 'r', encoding='utf-8') as f:
                data = json.load(f)
                results.append(data)
                
                # 统计错误类型
                for error in data.get('Details', {}).get('Errors', []):
                    error_code = extract_error_code(error)
                    if error_code:
                        error_summary[error_code] += 1
                
                # 统计警告类型
                for warning in data.get('Details', {}).get('Warnings', []):
                    warning_code = extract_error_code(warning)
                    if warning_code:
                        warning_summary[warning_code] += 1
    
    # 生成汇总报告
    generate_summary_report(results, error_summary, warning_summary)

def extract_error_code(message):
    """从编译消息中提取错误码"""
    import re
    match = re.search(r'(CS\d+)', message)
    return match.group(1) if match else None

def generate_summary_report(results, error_summary, warning_summary):
    """生成编译状态汇总报告"""
    
    total_projects = len(results)
    total_errors = sum(r['Errors'] for r in results)
    total_warnings = sum(r['Warnings'] for r in results)
    
    report = {
        'summary': {
            'total_projects': total_projects,
            'total_errors': total_errors,
            'total_warnings': total_warnings,
            'projects_with_errors': len([r for r in results if r['Errors'] > 0]),
            'projects_with_warnings': len([r for r in results if r['Warnings'] > 0])
        },
        'error_breakdown': dict(error_summary),
        'warning_breakdown': dict(warning_summary),
        'project_details': results
    }
    
    # 保存汇总报告
    with open('compilation-status-summary.json', 'w', encoding='utf-8') as f:
        json.dump(report, f, ensure_ascii=False, indent=2)
    
    # 生成Excel报告
    generate_excel_report(results, error_summary, warning_summary)

def generate_excel_report(results, error_summary, warning_summary):
    """生成Excel格式的详细报告"""
    
    # 项目汇总表
    df_projects = pd.DataFrame([
        {
            '项目名称': r['Project'],
            '项目路径': r['Path'],
            '编译错误数': r['Errors'],
            '编译警告数': r['Warnings'],
            '状态': '失败' if r['Errors'] > 0 else '成功'
        }
        for r in results
    ])
    
    # 错误类型汇总表
    df_errors = pd.DataFrame([
        {'错误类型': code, '出现次数': count, '优先级': get_error_priority(code)}
        for code, count in error_summary.items()
    ]).sort_values('出现次数', ascending=False)
    
    # 警告类型汇总表
    df_warnings = pd.DataFrame([
        {'警告类型': code, '出现次数': count, '建议处理': get_warning_suggestion(code)}
        for code, count in warning_summary.items()
    ]).sort_values('出现次数', ascending=False)
    
    # 保存Excel报告
    with pd.ExcelWriter('compilation-status-report.xlsx', engine='openpyxl') as writer:
        df_projects.to_excel(writer, sheet_name='项目汇总', index=False)
        df_errors.to_excel(writer, sheet_name='错误分析', index=False)
        df_warnings.to_excel(writer, sheet_name='警告分析', index=False)

def get_error_priority(error_code):
    """获取错误优先级"""
    high_priority = ['CS0246', 'CS0103', 'CS1061', 'MSB3644']
    medium_priority = ['CS0117', 'CS0029', 'CS0266']
    
    if error_code in high_priority:
        return '高'
    elif error_code in medium_priority:
        return '中'
    else:
        return '低'

def get_warning_suggestion(warning_code):
    """获取警告处理建议"""
    suggestions = {
        'CS8600': '添加空值检查',
        'CS8602': '使用空条件操作符',
        'CS8618': '在构造函数中初始化非空属性',
        'CS1998': '添加await或移除async',
        'CS0649': '初始化字段或使用属性',
        'CS0168': '移除未使用的变量'
    }
    
    return suggestions.get(warning_code, '请查阅官方文档')

if __name__ == '__main__':
    analyze_compilation_results()
    print("编译状态分析完成，请查看 compilation-status-report.xlsx")
```
