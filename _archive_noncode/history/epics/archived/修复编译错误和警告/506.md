```csharp
// tools/ComplexityAnalyzer.cs
using Microsoft.CodeAnalysis;
using Microsoft.CodeAnalysis.CSharp;
using Microsoft.CodeAnalysis.CSharp.Syntax;

public class ComplexityAnalyzer : CSharpSyntaxWalker
{
    private Dictionary<string, int> _methodComplexity = new();
    private string _currentMethod = string.Empty;
    private int _currentComplexity = 1;

    public override void VisitMethodDeclaration(MethodDeclarationSyntax node)
    {
        _currentMethod = GetMethodName(node);
        _currentComplexity = 1; // 基础复杂度
        
        base.VisitMethodDeclaration(node);
        
        _methodComplexity[_currentMethod] = _currentComplexity;
        
        // 复杂度警告
        if (_currentComplexity > 10)
        {
            Console.WriteLine($"高复杂度方法警告: {_currentMethod} (复杂度: {_currentComplexity})");
        }
    }

    public override void VisitIfStatement(IfStatementSyntax node)
    {
        _currentComplexity++;
        base.VisitIfStatement(node);
    }

    public override void VisitWhileStatement(WhileStatementSyntax node)
    {
        _currentComplexity++;
        base.VisitWhileStatement(node);
    }

    public override void VisitForStatement(ForStatementSyntax node)
    {
        _currentComplexity++;
        base.VisitForStatement(node);
    }

    public override void VisitSwitchStatement(SwitchStatementSyntax node)
    {
        _currentComplexity += node.Sections.Count;
        base.VisitSwitchStatement(node);
    }

    private string GetMethodName(MethodDeclarationSyntax method)
    {
        var className = method.Ancestors<ClassDeclarationSyntax>().FirstOrDefault()?.Identifier.Text ?? "Unknown";
        return $"{className}.{method.Identifier.Text}";
    }

    public Dictionary<string, int> GetComplexityReport() => _methodComplexity;
}
```

#### 指标2: 代码重复度 (Code Duplication)

```python
# scripts/duplication-detector.py
import ast
import hashlib
from collections import defaultdict
import os

class DuplicationDetector:
    def __init__(self, min_lines=5):
        self.min_lines = min_lines
        self.code_blocks = defaultdict(list)
        
    def analyze_file(self, file_path):
        """分析单个C#文件的代码重复"""
        
        with open(file_path, 'r', encoding='utf-8') as f:
            lines = f.readlines()
        
        # 滑动窗口检测重复代码块
        for i in range(len(lines) - self.min_lines + 1):
            block = ''.join(lines[i:i + self.min_lines])
            normalized_block = self.normalize_code(block)
            
            if normalized_block.strip():
                block_hash = hashlib.md5(normalized_block.encode()).hexdigest()
                self.code_blocks[block_hash].append({
                    'file': file_path,
                    'start_line': i + 1,
                    'end_line': i + self.min_lines,
                    'content': block
                })
    
    def normalize_code(self, code):
        """标准化代码，移除空格和注释"""
        import re
        
        # 移除注释
        code = re.sub(r'//.*$', '', code, flags=re.MULTILINE)
        code = re.sub(r'/\*.*?\*/', '', code, flags=re.DOTALL)
        
        # 标准化空格
        code = re.sub(r'\s+', ' ', code)
        
        return code.strip()
    
    def find_duplications(self):
        """查找重复代码块"""
        duplications = []
        
        for block_hash, occurrences in self.code_blocks.items():
            if len(occurrences) > 1:
                duplications.append({
                    'hash': block_hash,
                    'count': len(occurrences),
                    'occurrences': occurrences
                })
        
        return sorted(duplications, key=lambda x: x['count'], reverse=True)
    
    def generate_report(self):
        """生成重复代码报告"""
        duplications = self.find_duplications()
        
        report = {
            'summary': {
                'total_duplications': len(duplications),
                'total_lines_duplicated': sum(
                    d['count'] * self.min_lines for d in duplications
                )
            },
            'details': duplications
        }
        
        with open('code-duplication-report.json', 'w', encoding='utf-8') as f:
            json.dump(report, f, ensure_ascii=False, indent=2)
        
        return report

def main():
    detector = DuplicationDetector(min_lines=5)
    
    # 扫描所有C#文件
    for root, dirs, files in os.walk('.'):
        if 'obj' in root or 'bin' in root:
            continue
            
        for file in files:
            if file.endswith('.cs'):
                file_path = os.path.join(root, file)
                detector.analyze_file(file_path)
    
    report = detector.generate_report()
    print(f"代码重复分析完成: 发现 {report['summary']['total_duplications']} 处重复")

if __name__ == '__main__':
    main()
```

```csharp
// analyzers/UltraThinkQualityAnalyzer.cs
[DiagnosticAnalyzer(LanguageNames.CSharp)]
public class UltraThinkQualityAnalyzer : DiagnosticAnalyzer
{
    public static readonly DiagnosticDescriptor LongMethodRule = new(
        "UT001",
        "方法过长",
        "方法 '{0}' 包含 {1} 行代码，建议重构为较小的方法",
        "Design",
        DiagnosticSeverity.Warning,
        isEnabledByDefault: true);

    public static readonly DiagnosticDescriptor TooManyParametersRule = new(
        "UT002", 
        "参数过多",
        "方法 '{0}' 有 {1} 个参数，建议使用参数对象模式",
        "Design",
        DiagnosticSeverity.Warning,
        isEnabledByDefault: true);

    public static readonly DiagnosticDescriptor ServiceNamingRule = new(
        "UT003",
        "UltraThink服务命名规范",
        "服务类 '{0}' 应该以 'Service' 结尾并实现 IXxxService 接口",
        "Naming",
        DiagnosticSeverity.Warning,
        isEnabledByDefault: true);

    public override ImmutableArray<DiagnosticDescriptor> SupportedDiagnostics =>
        ImmutableArray.Create(LongMethodRule, TooManyParametersRule, ServiceNamingRule);

    public override void Initialize(AnalysisContext context)
    {
        context.EnableConcurrentExecution();
        context.ConfigureGeneratedCodeAnalysis(GeneratedCodeAnalysisFlags.None);
        context.RegisterSyntaxNodeAction(AnalyzeMethodDeclaration, SyntaxKind.MethodDeclaration);
        context.RegisterSyntaxNodeAction(AnalyzeClassDeclaration, SyntaxKind.ClassDeclaration);
    }

    private static void AnalyzeMethodDeclaration(SyntaxNodeAnalysisContext context)
    {
        var method = (MethodDeclarationSyntax)context.Node;
        
        // 检查方法长度
        var lineCount = method.GetLocation().GetLineSpan().EndLinePosition.Line -
                       method.GetLocation().GetLineSpan().StartLinePosition.Line + 1;
        
        if (lineCount > 50)
        {
            var diagnostic = Diagnostic.Create(
                LongMethodRule,
                method.Identifier.GetLocation(),
                method.Identifier.Text,
                lineCount);
            context.ReportDiagnostic(diagnostic);
        }
        
        // 检查参数数量
        if (method.ParameterList.Parameters.Count > 5)
        {
            var diagnostic = Diagnostic.Create(
                TooManyParametersRule,
                method.ParameterList.GetLocation(),
                method.Identifier.Text,
                method.ParameterList.Parameters.Count);
            context.ReportDiagnostic(diagnostic);
        }
    }

    private static void AnalyzeClassDeclaration(SyntaxNodeAnalysisContext context)
    {
        var classDecl = (ClassDeclarationSyntax)context.Node;
        var className = classDecl.Identifier.Text;
        
        // UltraThink架构规范检查
        if (className.Contains("Service") && !className.EndsWith("Service"))
        {
            var diagnostic = Diagnostic.Create(
                ServiceNamingRule,
                classDecl.Identifier.GetLocation(),
                className);
            context.ReportDiagnostic(diagnostic);
        }
    }
}
```

#### 工具2: 质量监控仪表板

```python
# tools/quality-dashboard.py
import json
import os
from datetime import datetime
import matplotlib.pyplot as plt
import seaborn as sns

class QualityDashboard:
    def __init__(self):
        self.metrics_history = []
        
    def collect_metrics(self):
        """收集各种质量指标"""
        
        metrics = {
            'timestamp': datetime.now().isoformat(),
            'compilation': self.get_compilation_metrics(),
            'complexity': self.get_complexity_metrics(),
            'duplication': self.get_duplication_metrics(),
            'coverage': self.get_coverage_metrics(),
            'violations': self.get_violation_metrics()
        }
        
        return metrics
    
    def get_compilation_metrics(self):
        """获取编译质量指标"""
        if os.path.exists('compilation-status-summary.json'):
            with open('compilation-status-summary.json', 'r') as f:
                data = json.load(f)
                return {
                    'total_projects': data['summary']['total_projects'],
                    'projects_with_errors': data['summary']['projects_with_errors'],
                    'projects_with_warnings': data['summary']['projects_with_warnings'],
                    'total_errors': data['summary']['total_errors'],
                    'total_warnings': data['summary']['total_warnings']
                }
        return {}
    
    def get_complexity_metrics(self):
        """获取复杂度指标"""
        if os.path.exists('complexity-report.json'):
            with open('complexity-report.json', 'r') as f:
                data = json.load(f)
                complexities = list(data.values())
                return {
                    'average_complexity': sum(complexities) / len(complexities) if complexities else 0,
                    'max_complexity': max(complexities) if complexities else 0,
                    'high_complexity_methods': len([c for c in complexities if c > 10])
                }
        return {}
    
    def get_duplication_metrics(self):
        """获取代码重复指标"""
        if os.path.exists('code-duplication-report.json'):
            with open('code-duplication-report.json', 'r') as f:
                data = json.load(f)
                return data['summary']
        return {}
    
    def get_coverage_metrics(self):
        """获取测试覆盖率指标"""
        if os.path.exists('CoverageReport/Summary.json'):
            with open('CoverageReport/Summary.json', 'r') as f:
                data = json.load(f)
                return {
                    'line_coverage': data['summary']['linecoverage'],
                    'branch_coverage': data['summary']['branchcoverage'],
                    'method_coverage': data['summary']['methodcoverage']
                }
        return {}
    
    def get_violation_metrics(self):
        """获取代码违规指标"""
        # 这里可以解析SonarQube、分析器输出等
        return {
            'critical_issues': 0,
            'major_issues': 0,
            'minor_issues': 0,
            'code_smells': 0
        }
    
    def save_metrics(self, metrics):
        """保存指标到历史记录"""
        history_file = 'quality-metrics-history.json'
        
        if os.path.exists(history_file):
            with open(history_file, 'r') as f:
                self.metrics_history = json.load(f)
        
        self.metrics_history.append(metrics)
        
        with open(history_file, 'w') as f:
            json.dump(self.metrics_history, f, indent=2)
    
    def generate_dashboard(self):
        """生成质量仪表板"""
        if not self.metrics_history:
            print("没有历史数据，无法生成仪表板")
            return
        
        fig, axes = plt.subplots(2, 2, figsize=(15, 10))
        fig.suptitle('代码质量监控仪表板', fontsize=16)
        
        # 编译错误趋势
        timestamps = [m['timestamp'] for m in self.metrics_history]
        errors = [m.get('compilation', {}).get('total_errors', 0) for m in self.metrics_history]
        warnings = [m.get('compilation', {}).get('total_warnings', 0) for m in self.metrics_history]
        
        axes[0, 0].plot(timestamps, errors, label='错误', color='red')
        axes[0, 0].plot(timestamps, warnings, label='警告', color='orange')
        axes[0, 0].set_title('编译问题趋势')
        axes[0, 0].legend()
        axes[0, 0].tick_params(axis='x', rotation=45)
        
        # 测试覆盖率趋势
        coverage = [m.get('coverage', {}).get('line_coverage', 0) for m in self.metrics_history]
        axes[0, 1].plot(timestamps, coverage, label='行覆盖率', color='green')
        axes[0, 1].set_title('测试覆盖率趋势')
        axes[0, 1].set_ylabel('覆盖率 (%)')
        axes[0, 1].legend()
        axes[0, 1].tick_params(axis='x', rotation=45)
        
        # 复杂度分布
        latest_complexity = self.metrics_history[-1].get('complexity', {})
        if latest_complexity:
            axes[1, 0].bar(['平均复杂度', '最大复杂度', '高复杂度方法数'], [
                latest_complexity.get('average_complexity', 0),
                latest_complexity.get('max_complexity', 0),
                latest_complexity.get('high_complexity_methods', 0)
            ])
            axes[1, 0].set_title('当前复杂度指标')
        
        # 代码重复情况
        duplication = [m.get('duplication', {}).get('total_duplications', 0) for m in self.metrics_history]
        axes[1, 1].plot(timestamps, duplication, label='重复代码块', color='purple')
        axes[1, 1].set_title('代码重复趋势')
        axes[1, 1].legend()
        axes[1, 1].tick_params(axis='x', rotation=45)
        
        plt.tight_layout()
        plt.savefig('quality-dashboard.png', dpi=300, bbox_inches='tight')
        plt.close()
        
        print("质量仪表板已生成: quality-dashboard.png")

def main():
    dashboard = QualityDashboard()
    
    # 收集当前指标
    current_metrics = dashboard.collect_metrics()
    dashboard.save_metrics(current_metrics)
    
    # 生成仪表板
    dashboard.generate_dashboard()
    
    # 输出当前质量概览
    print("当前代码质量概览:")
    print(f"  编译错误: {current_metrics.get('compilation', {}).get('total_errors', 'N/A')}")
    print(f"  编译警告: {current_metrics.get('compilation', {}).get('total_warnings', 'N/A')}")
    print(f"  测试覆盖率: {current_metrics.get('coverage', {}).get('line_coverage', 'N/A')}%")
    print(f"  平均复杂度: {current_metrics.get('complexity', {}).get('average_complexity', 'N/A')}")
    print(f"  代码重复: {current_metrics.get('duplication', {}).get('total_duplications', 'N/A')} 处")

if __name__ == '__main__':
    main()
```
