## 技术实现要点

### 现有功能改进
```csharp
// 优化现有HerbImportViewModel
public class HerbImportViewModel : ViewModelBase
{
    // 增强的数据验证
    private ValidationResult ValidateHerbData(HerbImportDto herb)
    {
        var result = new ValidationResult();
        
        // 名称重复检测
        if (await _herbModule.IsHerbNameExistsAsync(herb.Name))
            result.AddError("药材名称已存在");
            
        // 价格合理性检测
        if (herb.Price < 0 || herb.Price > 10000)
            result.AddError("药材价格超出合理范围");
            
        return result;
    }
    
    // 批量价格调整
    public async Task BatchUpdatePricesAsync(PriceUpdateCriteria criteria)
    {
        var selectedHerbs = GetSelectedHerbs();
        await _herbModule.BatchUpdatePricesAsync(selectedHerbs, criteria);
    }
}
```

### 新增批量管理功能
```csharp
public class HerbBatchManagementViewModel : ViewModelBase
{
    private readonly IHerbModule _herbModule;
    
    // 批量价格调整
    public async Task ExecutePriceAdjustmentAsync()
    {
        var criteria = new PriceAdjustmentCriteria
        {
            AdjustmentType = SelectedAdjustmentType, // 百分比或固定金额
            AdjustmentValue = AdjustmentValue,
            CategoryFilter = SelectedCategories
        };
        
        await _herbModule.BatchAdjustPricesAsync(criteria);
    }
    
    // 批量分类设置
    public async Task BatchSetCategoryAsync(string category)
    {
        var selectedHerbs = GetSelectedHerbs();
        await _herbModule.BatchUpdateCategoryAsync(selectedHerbs, category);
    }
}
```

public class EnhancedHerbValidator
{
    public ValidationResult ValidateImportData(List<HerbImportDto> herbs)
    {
        var result = new ValidationResult();
        
        // 1. 数据完整性检查
        ValidateRequiredFields(herbs, result);
        
        // 2. 数据格式检查
        ValidateDataFormats(herbs, result);
        
        // 3. 重复数据检查
        ValidateDuplicates(herbs, result);
        
        // 4. 价格合理性检查
        ValidatePriceRanges(herbs, result);
        
        return result;
    }
}
