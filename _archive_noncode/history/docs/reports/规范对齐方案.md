```csharp
public async Task<ServiceResult<bool>> ChangePasswordAsync(Guid userId, ChangePasswordDto dto)
{
    // 验证当前密码是否正确 - 安全检查的关键步骤
    var currentUser = await _userRepository.GetByIdAsync(userId);
    if (currentUser == null)
    {
        return ServiceResult<bool>.Failure("用户不存在");
    }

    // 使用BCrypt验证密码哈希 - 防止彩虹表攻击
    if (!BCrypt.Net.BCrypt.Verify(dto.CurrentPassword, currentUser.PasswordHash))
    {
        // 记录密码验证失败事件 - 用于安全审计
        _logger.LogWarning("用户 {UserId} 密码验证失败，可能存在恶意尝试", userId);
        return ServiceResult<bool>.Failure("当前密码错误");
    }

    // 生成新密码哈希 - 使用BCrypt自动加盐
    var newPasswordHash = BCrypt.Net.BCrypt.HashPassword(dto.NewPassword);
    currentUser.PasswordHash = newPasswordHash;
    currentUser.UpdateTime = DateTime.Now;

    await _userRepository.UpdateAsync(currentUser);
    return ServiceResult<bool>.Success(true, "密码修改成功");
}
```

```csharp
// ✅ 标准异常处理模式 - 基于ServiceResult<T>
public async Task<ServiceResult<UserDto>> GetByIdAsync(Guid id)
{
    try
    {
        // 参数验证
        if (id == Guid.Empty)
        {
            return ServiceResult<UserDto>.Failure("用户ID不能为空");
        }

        // 业务逻辑执行
        var user = await _userRepository.GetByIdAsync(id);
        if (user == null)
        {
            return ServiceResult<UserDto>.Failure("用户不存在");
        }

        var userDto = _mapper.Map<UserDto>(user);
        return ServiceResult<UserDto>.Success(userDto, "查询成功");
    }
    catch (DbException ex)
    {
        // 数据库异常 - 记录详细错误但返回用户友好信息
        _logger.LogError(ex, "查询用户 {UserId} 时发生数据库错误", id);
        return ServiceResult<UserDto>.Failure("数据访问异常，请稍后重试");
    }
    catch (Exception ex)
    {
        // 未预期异常 - 记录完整堆栈跟踪
        _logger.LogError(ex, "查询用户 {UserId} 时发生未预期错误", id);
        return ServiceResult<UserDto>.Failure("系统异常，请联系管理员");
    }
}
```

```csharp
            .Include(u => u.Role) // 显式加载相关数据
            .FirstOrDefaultAsync(u => u.Id == id)
            .ConfigureAwait(false);
    }

    public async Task<PagedResult<User>> GetPagedAsync(int pageIndex, int pageSize, string? searchTerm = null)
    {
        var query = _context.Users.AsQueryable();
        
        // 条件筛选
        if (!string.IsNullOrEmpty(searchTerm))
        {
            query = query.Where(u => u.UserName.Contains(searchTerm) || 
                                   u.Email.Contains(searchTerm));
        }
        
        var totalCount = await query.CountAsync().ConfigureAwait(false);
        
        var items = await query
            .OrderBy(u => u.UserName)
            .Skip((pageIndex - 1) * pageSize)
            .Take(pageSize)
            .ToListAsync()
            .ConfigureAwait(false);
            
        return new PagedResult<User>(items, totalCount, pageIndex, pageSize);
    }

    public async Task<User> CreateAsync(User user)
    {
        _context.Users.Add(user);
        await _context.SaveChangesAsync().ConfigureAwait(false);
        
        _logger.LogInformation("用户创建成功: {UserId}", user.Id);
        return user;
    }
}
```

```csharp
// ✅ 统一的枚举扩展方法
public static class EnumExtensions
{
    /// <summary>
    /// 获取枚举的中文描述
    /// </summary>
    /// <param name="enumValue">枚举值</param>
    /// <returns>中文描述，如果没有Description特性则返回枚举名称</returns>
    public static string ToChinese(this Enum enumValue)
    {
        var field = enumValue.GetType().GetField(enumValue.ToString());
        if (field == null) return enumValue.ToString();

        var attribute = Attribute.GetCustomAttribute(field, typeof(DescriptionAttribute)) as DescriptionAttribute;
        return attribute?.Description ?? enumValue.ToString();
    }

    /// <summary>
    /// 获取枚举的所有项及其中文描述
    /// </summary>
    /// <typeparam name="T">枚举类型</typeparam>
    /// <returns>枚举项和描述的字典</returns>
    public static Dictionary<T, string> GetDescriptions<T>() where T : Enum
    {
        return Enum.GetValues<T>()
            .ToDictionary(value => value, value => value.ToChinese());
    }

    /// <summary>
    /// 从中文描述获取枚举值
    /// </summary>
    /// <typeparam name="T">枚举类型</typeparam>
    /// <param name="description">中文描述</param>
    /// <returns>对应的枚举值，如果找不到则返回null</returns>
    public static T? FromChinese<T>(string description) where T : struct, Enum
    {
        var fields = typeof(T).GetFields();
        foreach (var field in fields)
        {
            if (field.GetCustomAttribute<DescriptionAttribute>()?.Description == description)
            {
                return (T)field.GetValue(null);
            }
        }
        return null;
    }
}

// ✅ 使用示例
public class UserViewModel
{
    public string StatusText => User.Status.ToChinese(); // "启用"
    
    public Dictionary<UserStatus, string> StatusOptions => 
        EnumExtensions.GetDescriptions<UserStatus>();
}
```
